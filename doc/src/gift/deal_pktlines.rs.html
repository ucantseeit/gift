5#[derive(Debug)]
6#[derive(PartialEq)]
7pub enum PktLine {
8    Data(Vec<u8>),  // 一条普通数据行,里面是去掉长度前缀后的原始负载
9    Flush,          // 0000
10    Delim,          // 0001(v1 用不到,但先占好位,做 v2 时省事)
11}
12
13// 用来检查读前四个字节的时候读到的是否规范
14// 读满 buf,但允许"一开始就是 EOF"这种干净结束。
15// 返回 Ok(true) = 读满了;Ok(false) = 一个字节没读到就 EOF。
16fn read_exact_or_eof<R: Read>(reader: &mut R, buf: &mut [u8]) -> io::Result<bool> {
17    let mut filled = 0;
18    while filled < buf.len() {
19        match reader.read(&mut buf[filled..]) {
20            Ok(0) => {
21                if filled == 0 {
22                    return Ok(false); // 干净 EOF
23                } else {// 已经读了一半(filled>0)又 Ok(0) → 长度前缀读到一半流就没了 → 损坏
24                    return Err(io::Error::new(
25                        io::ErrorKind::UnexpectedEof,
26                        "读长度前缀读到一半流就断了",
27                    ));
28                }
29            }
30            Ok(n) => filled += n,
31            Err(ref e) if e.kind() == io::ErrorKind::Interrupted => continue,
32            Err(e) => return Err(e),
33        }
34    }
35    Ok(true)
36}

38/// 从流里读取下一条 pkt-line。
39/// - Ok(Some(...)):正常读到一条
40/// - Ok(None):流已到底(EOF),没有更多内容
41/// - Err(...):底层读取出错,或字节不合法
42pub fn read_pkt_line<R: Read>(reader: &mut R) -> io::Result<Option<PktLine>> {
43    // 1. 先读 4 字节长度前缀。这里要区分两种情况:
44    //    - 干净的 EOF(一个字节都没读到)→ 正常结束,返回 None
45    //    - 读了一部分又断了 → 这是损坏的流,报错
46    let mut len_buf = [0u8; 4];
47    match read_exact_or_eof(reader, &mut len_buf)? {
48        false => return Ok(None), // 干净 EOF
49        true => {}
50    }
51
52    // 2. 4 个字符是十六进制 ASCII 文本,parse 成数字。
53    let len_str = std::str::from_utf8(&len_buf)
54        .map_err(|_| io::Error::new(io::ErrorKind::InvalidData, "长度前缀不是合法 ASCII"))?;
55    let len = u16::from_str_radix(len_str, 16)
56        .map_err(|_| io::Error::new(io::ErrorKind::InvalidData, "长度前缀不是合法十六进制"))?;
57
58    // 3. 先看是不是特殊包(特殊值 0/1/2)。
59    match len {
60        0 => return Ok(Some(PktLine::Flush)),
61        1 => return Ok(Some(PktLine::Delim)),
62        2 => return Ok(Some(PktLine::Flush)), // response-end,clone 用不到,先归一下
63        3 => return Err(io::Error::new(io::ErrorKind::InvalidData, "长度 3 非法")),
64        _ => {}
65    }
66
67    // 4. len >= 4。负载长度 = len - 4(关键:减掉前缀自己那 4 字节)。
68    let payload_len = (len - 4) as usize;
69    let mut payload = vec![0u8; payload_len];
70    reader.read_exact(&mut payload)?; // 必须读满,不能只 read 一次
71    Ok(Some(PktLine::Data(payload)))
72}

74/// 拿到一系列的pkt_line包vec
75pub fn read_all<R: Read>(reader: &mut R) -> io::Result<Vec<PktLine>> {
76    let mut lines = Vec::new();
77    while let Some(pkt) = read_pkt_line(reader)? {
78        lines.push(pkt);
79    }
80    Ok(lines)
81}

83// pkt-line 负载最大字节数：总长上限 0xFFF0 = 65520，减去 4 字节头。
84const MAX_PKTLINE_DATA: usize = 65516;
85
86/// 把一条 pkt-line 写进 writer。
87pub fn write_pkt_line(w: &mut impl Write, line: &PktLine) -> io::Result<()> {
88    match line {
89        PktLine::Flush => w.write_all(b"0000"),
90        PktLine::Delim => w.write_all(b"0001"),
91        PktLine::Data(payload) => {
92            if payload.len() > MAX_PKTLINE_DATA {
93                return Err(io::Error::new(
94                    io::ErrorKind::InvalidInput,
95                    "pkt-line payload too large",
96                ));
97            }
98            let len = payload.len() + 4;        // ★ 长度含这 4 字节前缀本身
99            write!(w, "{:04x}", len)?;           // 4 位小写十六进制、零填充
100            w.write_all(payload)
101        }
102    }
103}

106#[test]
107fn roundtrip() {
108    // 编码已知用例
109    let mut buf = Vec::new();
110    write_pkt_line(&mut buf, &PktLine::Data(b"a\n".to_vec())).unwrap();
111    assert_eq!(buf, b"0006a\n");                // 2 字节负载 + 4 = 6 → "0006"
112
113    let mut f = Vec::new();
114    write_pkt_line(&mut f, &PktLine::Flush).unwrap();
115    assert_eq!(f, b"0000");
116
117    // 编码再解码 == 原值
118    let mut s = Vec::new();
119    let original = PktLine::Data(b"want xyz\n".to_vec());
120    write_pkt_line(&mut s, &original).unwrap();
121    let decoded = read_pkt_line(&mut &s[..]).unwrap().unwrap();
122    assert_eq!(decoded, original);
123}

143    /// 现场用 git 造个仓库，返回 `upload-pack --advertise-refs` 的真字节。
144    /// 等价于：git init -b main demo; 写文件; add; commit; tag v1; upload-pack --advertise-refs demo
145    fn real_git_advert() -> Vec<u8> {
146        let dir = std::env::temp_dir().join(format!("pktline_{}", std::process::id()));
147        let _ = std::fs::remove_dir_all(&dir);
148        let env = [
149            ("GIT_AUTHOR_NAME", "t"), ("GIT_AUTHOR_EMAIL", "t@t"),
150            ("GIT_COMMITTER_NAME", "t"), ("GIT_COMMITTER_EMAIL", "t@t"),
151        ];
152        run(Command::new("git").args(["init", "-q"]).arg(&dir));
153        std::fs::write(dir.join("a.txt"), "hello\n").unwrap();
154        run(Command::new("git").arg("-C").arg(&dir).args(["add", "a.txt"]));
155        run(Command::new("git").arg("-C").arg(&dir).args(["commit", "-q", "-m", "first"]).envs(env));
156        run(Command::new("git").arg("-C").arg(&dir).args(["tag", "v1"]).envs(env));
157        let out = Command::new("git")
158            .args(["upload-pack", "--advertise-refs"]).arg(&dir)
159            .output().expect("upload-pack 失败");
160        assert!(out.status.success());
161        let _ = std::fs::remove_dir_all(&dir);
162        out.stdout
163    }

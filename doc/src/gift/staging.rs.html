12/// 将若干工作区路径暂存：对每个叶子对象计算 blob、写入 `git_abs/objects`，再更新 `git_abs/index`。
13///
14/// # 参数
15///
16/// - **`git_abs`**：Git 目录（内含 `objects/`、`index`），可为 `.git`、`.gift` 等。
17/// - **`worktree`**：工作区根目录；index 中的路径为其相对路径（经 [`index::index_path_bytes`]）。
18/// - **`inputs`**：待暂存路径列表。
19/// - **`recursive_dirs`**：为 `true` 时递归展开目录内的文件与符号链接；为 `false` 时若某条路径为目录则返回错误。
20///
21/// # 契约（调用方负责保证）
22/// - `git_abs`、`worktree`、`inputs` 中每条路径均为已 `canonicalize` 的绝对路径。
23/// - 每条 input 均落在 `worktree` 下且不在 `git_abs` 下。
24/// - 如需从用户原始输入（相对路径、未验证）构造上述路径，请先调用 [`resolve_stage_inputs`]。
25///
26/// # 说明
27/// - 不会遍历进入 `git_abs` 内部；叶子若落在 `git_abs` 下会报错。
28/// - 暂未实现 `.gitignore`。
29/// - 暂未实现同步删除工作区文件的操作。
30pub fn stage_paths(
31    git_abs: &Path,
32    worktree: &Path,
33    inputs: &[PathBuf],
34    recursive_dirs: bool,
35) -> Result<()> {
36    if inputs.is_empty() {
37        bail!("no paths to stage");
38    }
39
40    // 加载.gitignore规则
41    let ignore_rules = ignore::IgnoreRules::load(worktree)
42        .context("load .gitignore")?;
43
44    // 收集所有的"叶子文件"
45    let mut leaves = BTreeSet::<PathBuf>::new();
46    for root in inputs {
47        collect_leaves(root, git_abs, recursive_dirs, &mut leaves)?;
48    }
49
50    
51    let index_path = git_abs.join("index");
52    let mut index = if index_path.exists() {
53        index_file::parse_index_file(&index_path).with_context(|| {
54            format!("parse index {}", index_path.display())
55        })?
56    } else {
57        IndexFile::empty(2)
58    };
59
60    for leaf in &leaves {
61        ensure_not_inside_git_abs(leaf, &git_abs)?;
62
63        let path_bytes = index_file::index_path_bytes(worktree, leaf)?;
64        // 使用 BTreeMap 的 contains_key 方法检查是否已追踪
65        let is_tracked = index.get_entry(&path_bytes).is_some();
66        
67        // 核心逻辑：只有未追踪的文件才检查是否被忽略
68        let ignored = !is_tracked && ignore_rules.is_ignored(leaf);
69        
70        if ignored {
71            continue;  // 跳过被忽略且未追踪的文件
72        }
73
74        let (sha, blob_content) = object::hash_object(leaf)?;
75        object::write_hash_object(&git_abs, &sha, &blob_content)?;
76
77        let md = fs::symlink_metadata(leaf)?;
78        //let path_bytes = index_file::index_path_bytes(&worktree, leaf)?;
79        index_file::add_index(
80            &mut index,
81            &md,
82            path_bytes,
83            sha,
84        )
85        .with_context(|| format!("add_index {}", leaf.display()))?;
86    }
87
88    index_file::write_index_file(&index_path, &index).with_context(|| {
89        format!("write_index_file {}", index_path.display())
90    })?;
91
92    Ok(())
93}

95/// 将用户原始输入路径规范化为 [`stage_paths`] 所要求的 canon 绝对路径列表。
96///
97/// 相对路径相对于**进程当前工作目录**解析，然后 canonicalize，再校验落在 `worktree` 内且不在 `git_abs` 内。
98pub fn resolve_stage_inputs(
99    inputs: &[PathBuf],
100    worktree: &Path,
101    git_abs: &Path,
102) -> Result<Vec<PathBuf>> {
103    let cwd = std::env::current_dir()?;
104    let mut resolved = Vec::new();
105    for input in inputs {
106        let p = resolve_under_worktree(input, &cwd, worktree)?;
107        ensure_not_inside_git_abs(&p, git_abs)?;
108        resolved.push(p);
109    }
110    Ok(resolved)
111}

113/// 将用户输入路径解析为落在 `worktree_canon` 下的规范绝对路径。
114fn resolve_under_worktree(
115    input: &Path,
116    cwd: &Path,
117    worktree_canon: &Path,
118) -> Result<PathBuf> {
119    let joined = if input.is_absolute() {
120        input.to_path_buf()
121    } else {
122        cwd.join(input)
123    };
124    
125    let canon =
126        fs::canonicalize(&joined).with_context(|| {
127            format!("canonicalize {}", joined.display())
128        })?;
129    canon.strip_prefix(worktree_canon).with_context(|| {
130        format!(
131            "{} is not under work tree {}",
132            canon.display(),
133            worktree_canon.display()
134        )
135    })?;
136    Ok(canon)
137}

139/// 确保path不在git_abs以内
140fn ensure_not_inside_git_abs(path: &Path, git_abs: &Path) -> Result<()> {
141    let path = fs::canonicalize(path)?;
142    if path.starts_with(git_abs) {
143        bail!(
144            "path {} lies inside git directory {}",
145            path.display(),
146            git_abs.display()
147        );
148    }
149    Ok(())
150}

152/// 若为文件或符号链接则加入 `out`；若为目录则按需递归。
153fn collect_leaves(
154    path: &Path,
155    git_abs: &Path,
156    recursive_dir: bool,
157    out: &mut BTreeSet<PathBuf>,
158) -> Result<()> {
159    let md = fs::symlink_metadata(path)
160        .with_context(|| format!("symlink_metadata {}", path.display()))?;
161
162    if md.file_type().is_symlink() || md.is_file() {
163        out.insert(path.to_path_buf());
164        return Ok(());
165    }
166
167    if md.is_dir() {
168        let dir_canon =
169            fs::canonicalize(path).with_context(|| format!("canonicalize {}", path.display()))?;
170        if dir_canon.starts_with(git_abs) {
171            bail!(
172                "directory {} is inside git metadata dir {}",
173                path.display(),
174                git_abs.display()
175            );
176        }
177
178        if !recursive_dir {
179            bail!(
180                "{} is a directory; pass recursive_dirs = true",
181                path.display()
182            );
183        }
184
185        for entry in fs::read_dir(path).with_context(|| format!("read_dir {}", path.display()))?
186        {
187            let entry = entry.with_context(|| format!("read_dir entry {}", path.display()))?;
188            let child_path = entry.path();
189            let name = entry.file_name();
190            if name == ".git" || name == ".gift" {
191                continue;
192            }
193
194            let child_canon = fs::canonicalize(&child_path).with_context(|| {
195                format!("canonicalize {}", child_path.display())
196            })?;
197            if child_canon.starts_with(git_abs) {
198                continue;
199            }
200
201            collect_leaves(&child_path, git_abs, recursive_dir, out)?;
202        }
203        return Ok(());
204    }
205
206    bail!("unsupported type at {}", path.display());
207}

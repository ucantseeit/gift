3//! 复用既有 git 能力：`init` / `stage_paths` / `commit` / `branch` / `checkout` / `log`；
4//! 新增的只是 `.giftai` 仓库发现、DeepSeek 客户端、messages 组装/选择的编排。

24/// 新仓库的默认系统提示词（写入 `chat/0001-system.txt`）。
25const DEFAULT_SYSTEM_PROMPT: &str = "你是一个有用的 AI 助手，请用简洁清晰的中文回答。";

27/// commit message 概述的最大字符数。
28const SUMMARY_MAX_CHARS: usize = 50;

30#[derive(Parser, Debug)]
31#[command(
32    name = "giftai",
33    about = "基于 git 机制的 AI 对话上下文管理器",
34    long_about = None,
35)]
36struct Cli {
37    #[command(subcommand)]
38    command: Command,
39}

41#[derive(Subcommand, Debug)]
42enum Command {
43    /// 初始化对话仓库（建 .giftai + chat/ + 系统提示词）
44    Init {
45        /// 仓库根目录，缺省为当前目录
46        root: Option<PathBuf>,
47    },
48    /// 发起一轮对话：组装上下文 → 调 DeepSeek → 记录为新节点
49    Ask {
50        /// 只发送选中的轮（轮序号，如 `2,7`）；缺省发送全量快照
51        #[arg(long, value_delimiter = ',')]
52        select: Vec<String>,
53        /// 你的问题
54        question: String,
55    },
56    /// 预览本次将实际发送的 messages，不调用 API
57    Context {
58        /// 同 `ask --select`：只预览选中的轮
59        #[arg(long, value_delimiter = ',')]
60        select: Vec<String>,
61    },
62    /// 沿当前链查看历史：从根到 HEAD 按轮号 + 摘要列出（线性视图）
63    Log,
64    /// 跨所有分支画出整棵对话 DAG（带轮号、摘要、分支名的树状视图）
65    Graph,
66    /// 给当前方向命名（转调 git branch）
67    Branch {
68        /// 分支名
69        name: String,
70    },
71    /// 把另一条对话线合并进当前线（时间交错重编号 + 双 parent）
72    Merge {
73        /// 要并入的分支名
74        branch: String,
75    },
76    /// 切到某节点/方向（转调 git checkout）
77    Checkout {
78        /// 节点 OID 或分支名
79        target: String,
80    },
81}

83/// giftai CLI 入口：解析参数并分发到对应子命令。
84pub fn run() -> Result<()> {
85    let cli = Cli::parse();
86    match cli.command {
87        Command::Init { root } => run_init(root),
88        Command::Ask { select, question } => run_ask(&select, &question),
89        Command::Context { select } => run_context(&select),
90        Command::Log => run_log(),
91        Command::Graph => run_graph(),
92        Command::Branch { name } => run_branch(&name),
93        Command::Merge { branch } => run_merge(&branch),
94        Command::Checkout { target } => run_checkout(&target),
95    }
96}

98/// `giftai init [root]`：建 `.giftai` 元数据 + `chat/` 工作区 + 系统提示词；不立即提交。
99fn run_init(root: Option<PathBuf>) -> Result<()> {
100    let root = match root {
101        Some(p) => p,
102        None => std::env::current_dir().context("read current dir")?,
103    };
104    // root 可能尚不存在（用户指定新目录）→ 先建出来，再规范化为绝对路径。
105    fs::create_dir_all(&root).with_context(|| format!("create root {}", root.display()))?;
106    let root = fs::canonicalize(&root).with_context(|| format!("canonicalize {}", root.display()))?;
107
108    let git_abs = root.join(".giftai");
109    let worktree = root.join("chat");
110
111    if git_abs.exists() {
112        bail!("已是 giftai 仓库：{}", git_abs.display());
113    }
114
115    crate::init(&git_abs).map_err(|e| anyhow::anyhow!("init {}: {e}", git_abs.display()))?;
116    fs::create_dir_all(&worktree).with_context(|| format!("create chat {}", worktree.display()))?;
117
118    let sys_file = worktree.join(message_filename(1, Role::System));
119    fs::write(&sys_file, DEFAULT_SYSTEM_PROMPT)
120        .with_context(|| format!("write {}", sys_file.display()))?;
121
122    println!("已初始化 giftai 仓库：");
123    println!("  元数据 : {}", git_abs.display());
124    println!("  工作区 : {}", worktree.display());
125    println!("  系统提示词已写入 {}", sys_file.display());
126    println!("现在可以 `giftai ask \"你的问题\"` 开始对话（需设置 DEEPSEEK_API_KEY）。");
127    Ok(())
128}

130/// `giftai ask`：组装（可过滤）上下文 → 追加问题 → 调模型 → 写两条消息 → 暂存 + 提交。
131fn run_ask(select: &[String], question: &str) -> Result<()> {
132    let repo = discover_chat_repo()?;
133    let rounds = parse_select(select)?;
134
135    // 1. 组装本轮要发送的 messages：全量或按整轮过滤；末尾追加新问题。
136    let full = load_messages(&repo.worktree)?;
137    let mut to_send = if rounds.is_empty() {
138        full
139    } else {
140        select_rounds(&full, &rounds)?
141    };
142    to_send.push(Message::user(question));
143
144    // 2. 唯一的网络调用：失败则直接返回，不写文件、不提交。
145    let client = LlmClient::from_env()?;
146    let reply = client.chat(&to_send).context("调用 LLM 失败")?;
147
148    // 3. 写两条消息文件（问题在前、回答在后），保持完整快照。
149    let seq = next_seq(&repo.worktree)?;
150    let user_file = repo.worktree.join(message_filename(seq, Role::User));
151    let assistant_file = repo.worktree.join(message_filename(seq + 1, Role::Assistant));
152    fs::write(&user_file, question).with_context(|| format!("write {}", user_file.display()))?;
153    fs::write(&assistant_file, &reply)
154        .with_context(|| format!("write {}", assistant_file.display()))?;
155
156    // 4. 递归暂存整个工作区（首轮会一并纳入 0001-system.txt）。
157    let inputs = vec![repo.worktree.clone()];
158    let resolved = resolve_stage_inputs(&inputs, &repo.worktree, &repo.git_abs)?;
159    stage_paths(&repo.git_abs, &repo.worktree, &resolved, true)?;
160
161    // 5. 提交：parent 恒为 HEAD、tree 恒为完整快照（不受 --select 影响）。
162    //    概述作 message；若本轮做了过滤，把选中的轮号记进 message 末尾的 trailer
163    //    （trailer 本质是 message 正文，不走 CommitObject::trailing_headers）。
164    let (author, committer) = identities_from_git_env()?;
165    let message = build_commit_message(question, &rounds);
166    let new_oid = commit(&repo.worktree, &repo.git_abs, author, committer, Some(message))?;
167
168    println!("● {}  {}", new_oid.to_string(), summarize(question));
169    if !rounds.is_empty() {
170        println!("  （本轮仅基于第 {} 轮上下文生成）", join_rounds(&rounds));
171    }
172    println!("\n{reply}");
173    Ok(())
174}

176/// `giftai context`：组装（可过滤）上下文并打印，不追加问题、不调 API、不提交。
177fn run_context(select: &[String]) -> Result<()> {
178    let repo = discover_chat_repo()?;
179    let rounds = parse_select(select)?;
180
181    let full = load_messages(&repo.worktree)?;
182    let msgs = if rounds.is_empty() {
183        full
184    } else {
185        select_rounds(&full, &rounds)?
186    };
187
188    if rounds.is_empty() {
189        println!("将发送全量上下文，共 {} 条消息：\n", msgs.len());
190    } else {
191        println!(
192            "将发送第 {} 轮（含 system），共 {} 条消息：\n",
193            join_rounds(&rounds),
194            msgs.len()
195        );
196    }
197    for m in &msgs {
198        println!("── {} ──", m.role.keyword());
199        println!("{}\n", m.content);
200    }
201    Ok(())
202}

225fn run_checkout(target: &str) -> Result<()> {
226    let repo = discover_chat_repo()?;
227    let target: CheckoutTarget = target
228        .parse()
229        .map_err(|e| anyhow::anyhow!("解析 checkout 目标失败：{e}"))?;
230    checkout(&repo.worktree, &repo.git_abs, target)
231}

233/// 把 `--select` 的字符串解析为 1-based 轮号（demo 仅支持轮号，不支持 OID）。
234fn parse_select(select: &[String]) -> Result<Vec<u32>> {
235    let mut rounds = Vec::with_capacity(select.len());
236    for s in select {
237        let s = s.trim();
238        if s.is_empty() {
239            continue;
240        }
241        let n: u32 = s
242            .parse()
243            .with_context(|| format!("--select 只接受轮序号（正整数），无法解析：{s}"))?;
244        rounds.push(n);
245    }
246    Ok(rounds)
247}

249/// 构造 commit message：概述 +（若有过滤）trailer。
250fn build_commit_message(question: &str, rounds: &[u32]) -> String {
251    let summary = summarize(question);
252    if rounds.is_empty() {
253        summary
254    } else {
255        // trailer 与正文之间留空行，符合 git trailer 约定（位于 message 正文末尾）。
256        format!("{summary}\n\nGiftai-context: rounds {}", join_rounds(rounds))
257    }
258}

260/// 概述：取问题首行、去首尾空白、按字符截断到 [`SUMMARY_MAX_CHARS`]。
261fn summarize(question: &str) -> String {
262    let first_line = question.lines().next().unwrap_or("").trim();
263    let truncated: String = first_line.chars().take(SUMMARY_MAX_CHARS).collect();
264    if truncated.chars().count() < first_line.chars().count() {
265        format!("{truncated}…")
266    } else {
267        truncated
268    }
269}

271/// 把轮号列表格式化为 `1, 3` 形式。
272fn join_rounds(rounds: &[u32]) -> String {
273    rounds.iter().map(u32::to_string).collect::<Vec<_>>().join(", ")

1//! giftai 的「对话历史」视图：把 commit DAG 翻译成对人友好的**轮号**展示。
2//!
3//! 设计动机:用户做 `ask --select N` 时,凭记忆很难把「轮号」对上「具体聊了什么」,
4//! 更难看清分叉/祖先关系。因此本模块把轮号**自带说明地**打印出来,让选择从
5//! 「靠记忆」变成「查表」:
6//! - [`log`]:沿当前 HEAD 这一条链,从根到 HEAD 给每个节点标 `轮N` + 摘要(线性视图);
7//! - [`graph`]:跨所有分支把整棵 DAG 画成树,每个节点同样标 `轮N` + 摘要 + 分支名(全局视图)。
8//!
9//! 轮号约定(与 [`crate::ai::messages::select_rounds`] 完全一致):
10//! 每次 `ask` = 一个 commit = 新增一对 (user, assistant) = **一轮**;根 commit(首次
11//! 提问)= 轮 1,沿 parent 链每深一层轮号 +1。所以「`log`/`graph` 里看到的轮号」
12//! 就是「`--select` 能输入的轮号」,二者必然自洽。
13//!
14//! 轮号是**相对所在链**的:checkout 到别的分支后会按那条链重新计数——但因为显示与
15//! 选择都基于「当前节点这条链」算,不会出现对不上的情况。
16//!
17//! 本模块只读 git 对象与 refs,不写任何东西;不污染通用的 [`crate::log`](那是 git 复刻的
18//! 通用实现,gift / giftai 共用),giftai 专属的轮号语义只活在这里。
19

39/// `giftai log`:**读当前节点的快照**(`chat/` 工作区文件),按轮列出对话(轮号 + 问题 → 回答摘要)。
40///
41/// 数据来源刻意选「快照」而非「commit 链」:每轮 = 快照里的一对 (user, assistant),轮号 = 第几对——
42/// 这与 [`crate::ai::messages::select_rounds`] 是**同一批文件、同一种配对口径**,所以 `log` 显示的轮号
43/// 永远等于 `--select` 能输入的轮号。即便将来有 merge(一个合并 commit 吸收多轮、还可能插在中间,
44/// 使「commit 深度 ≠ 轮数」),快照口径也始终正确。
45///
46/// 结构信息(节点 OID、分支、分叉)不在这里展示——那是 [`graph`] 的职责;本视图仅在顶部留一行
47/// `当前节点 <oid> (分支)` 作定位。输出形如:
48/// ```text
49/// 当前节点 8290846b (master)   共 2 轮
50/// 轮1  记住一个数字：42         → 好的
51/// 轮2  我刚才让你记的数字是多少   → 42
52/// ```
53///
54/// - `worktree`:`chat/` 工作区(快照所在);
55/// - `git_abs`:仅用于读 HEAD,拼顶部那行定位信息(读不到时静默省略)。
56pub fn log(worktree: &Path, git_abs: &Path) -> Result<()> {
57    let messages = load_messages(worktree).context("读取对话快照失败")?;
58
59    // 开头连续的 system 为前缀,其余按 (user, assistant) 成对即「轮」。
60    let n_system = messages.iter().take_while(|m| m.role == Role::System).count();
61    let rounds = &messages[n_system..];
62    let total = rounds.len().div_ceil(2);
63
64    if total == 0 {
65        println!("(还没有任何对话轮次,先用 `giftai ask \"...\"` 开始)");
66        return Ok(());
67    }
68
69    // 顶部定位行:当前节点短 OID + 分支(都是尽力而为,失败就省略对应部分)。
70    let head = Head::read(git_abs).ok();
71    let node = head
72        .as_ref()
73        .and_then(|h| h.current_commit(git_abs).ok())
74        .map(|oid| short_hex(&oid));
75    let branch = head.as_ref().and_then(current_branch_name);
76    let mut header = String::from("当前节点 ");
77    header.push_str(node.as_deref().unwrap_or("(未提交)"));
78    if let Some(name) = &branch {
79        header.push_str(&format!(" ({name})"));
80    }
81    header.push_str(&format!("   共 {total} 轮"));
82    println!("{header}");
83
84    // 逐对打印:轮号 + 问题摘要 →（若有）回答摘要。
85    for (i, pair) in rounds.chunks(2).enumerate() {
86        let round = i + 1;
87        let question = line_summary(&pair[0].content);
88        match pair.get(1) {
89            Some(answer) => {
90                println!("轮{round}  {question}  → {}", line_summary(&answer.content))
91            }
92            // 配对里缺回答(理论上不该出现)——如实展示,不假装完整。
93            None => println!("轮{round}  {question}  → (无回答)"),
94        }
95    }
96    Ok(())
97}

99/// 取一段内容的首行非空文本,去空白并按 [`LINE_SUMMARY_MAX`] 字符截断(超出补 `…`)。
100fn line_summary(content: &str) -> String {
101    let first = content.lines().map(str::trim).find(|l| !l.is_empty()).unwrap_or("");
102    let truncated: String = first.chars().take(LINE_SUMMARY_MAX).collect();
103    if truncated.chars().count() < first.chars().count() {
104        format!("{truncated}…")
105    } else {
106        truncated
107    }
108}

110/// `giftai graph`:跨所有分支 + HEAD,把整棵对话 DAG 画成缩进树(`git log --graph --all` 的简化版)。
111///
112/// 按 first-parent 建树:`ask` 产生的单 parent 节点是 `● 轮N`(N = 该节点快照的轮数);merge
113/// 产生的多 parent 节点是 `◆ 合并`,并用 `⮌ ← <分支/oid>` 标出被并入的另一侧(不主动展开第二
114/// parent 的历史)。轮号用「快照轮数」而非链深度——合并后深度 ≠ 轮数,只有快照口径与 `--select`
115/// 一致。输出形如:
116/// ```text
117/// └── ● 轮1  0bc67959  记住一个数字：42
118///     ├── ● 轮2  5eb409a5  形容今天天气  [perf]
119///     └── ● 轮2  6f1062c8  数字是多少
120///         └── ◆ 合并  15735bc9  Merge perf into master  [master (当前)]  ⮌ ← perf
121/// ```
122pub fn graph(git_abs: &Path) -> Result<()> {
123    let head = Head::read(git_abs).context("读取 HEAD 失败")?;
124
125    // 1. 收集所有「树梢」:每个分支 tip,外加 detached HEAD 直接指向的 commit。
126    let branches = list_branches(git_abs)?;
127    let mut tips: Vec<ObjectSha> = branches.iter().map(|(_, oid)| oid.clone()).collect();
128    let detached = match &head {
129        Head::TargetCommit(oid) => Some(oid.clone()),
130        Head::TargetBranch(_) => None,
131    };
132    if let Some(oid) = &detached {
133        tips.push(oid.clone());
134    }
135
136    if tips.is_empty() {
137        println!("(还没有任何对话轮次,先用 `giftai ask \"...\"` 开始)");
138        return Ok(());
139    }
140
141    // 2. 从所有 tip 出发,沿**全部** parent 回溯,把可达的 commit 全读进内存(按 hex 去重)。
142    let mut commits: HashMap<String, CommitObject> = HashMap::new();
143    let mut stack = tips.clone();
144    while let Some(oid) = stack.pop() {
145        let hex = oid.to_string();
146        if commits.contains_key(&hex) {
147            continue;
148        }
149        let commit = CommitObject::read_loose_commit(git_abs, &hex)
150            .with_context(|| format!("读取 commit {hex}"))?;
151        for parent in &commit.parents {
152            stack.push(parent.clone());
153        }
154        commits.insert(hex, commit);
155    }
156
157    // 3. 建「父 → 子」邻接表,并找出根(无 parent 的节点)。
158    let mut children: HashMap<String, Vec<String>> = HashMap::new();
159    let mut roots: Vec<String> = Vec::new();
160    for (hex, commit) in &commits {
161        if commit.parents.is_empty() {
162            roots.push(hex.clone());
163        } else {
164            // first-parent 决定它在树里的归属与轮号(与 log 的链一致)。
165            let parent_hex = commit.parents[0].to_string();
166            children.entry(parent_hex).or_default().push(hex.clone());
167        }
168    }
169
170    // 4. 预备「节点 → 分支标签」映射,以及 HEAD 当前所在分支,用于渲染时标注。
171    let mut labels: HashMap<String, Vec<String>> = HashMap::new();
172    for (name, oid) in &branches {
173        labels.entry(oid.to_string()).or_default().push(name.clone());
174    }
175    let current_branch = current_branch_name(&head);
176
177    // 5. 预算每个节点的「快照轮数」(= 其 tree 里 NNNN-user 文件数),作为展示轮号。
178    //    用快照口径而非链深度:合并后深度 ≠ 轮数,只有快照数与 select_rounds / --select 对得上。
179    let mut rounds: HashMap<String, u32> = HashMap::new();
180    for (hex, commit) in &commits {
181        rounds.insert(hex.clone(), snapshot_rounds(git_abs, &commit.tree)?);
182    }
183
184    // 6. 从每个根递归渲染(根之间、子节点之间都按时间→hex 排序,输出稳定)。
185    sort_by_time(&mut roots, &commits);
186    let ctx = RenderCtx {
187        commits: &commits,
188        children: &children,
189        labels: &labels,
190        rounds: &rounds,
191        current_branch: &current_branch,
192        detached: &detached,
193    };
194    for (i, root) in roots.iter().enumerate() {
195        render_subtree(root, "", i + 1 == roots.len(), &ctx);
196    }
197    Ok(())
198}

202/// 渲染递归过程中只读的共享上下文,避免在递归签名里拖一长串参数。
203struct RenderCtx<'a> {
204    commits: &'a HashMap<String, CommitObject>,
205    children: &'a HashMap<String, Vec<String>>,
206    labels: &'a HashMap<String, Vec<String>>,
207    rounds: &'a HashMap<String, u32>,
208    current_branch: &'a Option<String>,
209    detached: &'a Option<ObjectSha>,
210}
211
212/// 递归打印以 `hex` 为根的子树。
213///
214/// - `prefix`:本行之前的缩进/竖线前缀(由上层根据「是否最后一个兄弟」累积而成);
215/// - `is_last`:本节点是否其父的最后一个子节点(决定用 `└──` 还是 `├──`)。
216///
217/// 普通节点渲染为 `● 轮N`(N 取自快照轮数);合并节点(≥2 parent)渲染为 `◆ 合并` 且不标轮号
218/// (它吸收并重排了多轮,标单个号会误导),改用 `⮌ ← <other parents>` 指出被并入的另一侧。
219fn render_subtree(hex: &str, prefix: &str, is_last: bool, ctx: &RenderCtx) {
220    let connector = if is_last { "└── " } else { "├── " };
221    let commit = &ctx.commits[hex];
222    let (summary, ctx_note) = commit_summary(&commit.message);
223    let is_merge = commit.parents.len() >= 2;
224
225    let marker = if is_merge {
226        "◆ 合并".to_string()
227    } else {
228        format!("● 轮{}", ctx.rounds.get(hex).copied().unwrap_or(0))
229    };
230    let mut line = format!("{prefix}{connector}{marker}  {}  {summary}", short_hex_str(hex));
231    if let Some(tags) = render_labels(hex, ctx) {
232        line.push_str(&format!("  {tags}"));
233    }
234    if is_merge {
235        line.push_str(&format!("  ⮌ ← {}", merge_sources(commit, ctx)));
236    }
237    if let Some(note) = ctx_note {
238        line.push_str(&format!("  ({note})"));
239    }
240    println!("{line}");
241
242    // 子节点的前缀:父若是最后一个兄弟,其下方用空白续接;否则用竖线 `│` 续接。
243    let child_prefix = format!("{prefix}{}", if is_last { "    " } else { "│   " });
244    let mut kids = ctx.children.get(hex).cloned().unwrap_or_default();
245    sort_by_time(&mut kids, ctx.commits);
246    for (i, kid) in kids.iter().enumerate() {
247        render_subtree(kid, &child_prefix, i + 1 == kids.len(), ctx);
248    }
249}

251/// 合并节点的「被并入侧」描述:取第 2 个及以后的 parent,有分支名用分支名,否则用短 oid。
252fn merge_sources(commit: &CommitObject, ctx: &RenderCtx) -> String {
253    commit.parents[1..]
254        .iter()
255        .map(|p| {
256            let ph = p.to_string();
257            match ctx.labels.get(&ph).and_then(|names| names.first()) {
258                Some(name) => name.clone(),
259                None => short_hex_str(&ph),
260            }
261        })
262        .collect::<Vec<_>>()
263        .join(", ")
264}

266/// 拼出某节点的标签串,如 `[master (当前), perf]` 或游离 HEAD 的 `[HEAD 游离]`;无标签返回 `None`。
267fn render_labels(hex: &str, ctx: &RenderCtx) -> Option<String> {
268    let mut tags: Vec<String> = Vec::new();
269    if let Some(names) = ctx.labels.get(hex) {
270        for name in names {
271            if ctx.current_branch.as_deref() == Some(name) {
272                tags.push(format!("{name} (当前)"));
273            } else {
274                tags.push(name.clone());
275            }
276        }
277    }
278    if ctx.detached.as_ref().map(|o| o.to_string()).as_deref() == Some(hex) {
279        tags.push("HEAD 游离".to_string());
280    }
281    if tags.is_empty() {
282        None
283    } else {
284        Some(format!("[{}]", tags.join(", ")))
285    }
286}

290/// 列出 `refs/heads/` 下所有分支,返回 `(分支名, tip OID)`。
291///
292/// 递归处理嵌套目录(如 `refs/heads/feature/x` → 分支名 `feature/x`);`refs/heads` 不存在时返回空。
293fn list_branches(git_abs: &Path) -> Result<Vec<(String, ObjectSha)>> {
294    let heads_dir = git_abs.join("refs").join("heads");
295    let mut out = Vec::new();
296    if heads_dir.is_dir() {
297        collect_refs(git_abs, &heads_dir, &heads_dir, &mut out)?;
298    }
299    out.sort_by(|a, b| a.0.cmp(&b.0));
300    Ok(out)
301}
302
303/// 递归遍历 `dir`,把每个 ref 文件解析成 `(相对 heads_root 的分支名, OID)` 收进 `out`。
304fn collect_refs(
305    git_abs: &Path,
306    heads_root: &Path,
307    dir: &Path,
308    out: &mut Vec<(String, ObjectSha)>,
309) -> Result<()> {
310    for ent in fs::read_dir(dir).with_context(|| format!("读取 {}", dir.display()))? {
311        let path = ent?.path();
312        if path.is_dir() {
313            collect_refs(git_abs, heads_root, &path, out)?;
314        } else {
315            // 分支名 = 相对 refs/heads 的路径(分隔符归一为 `/`,与 git ref 命名一致)。
316            let name = path
317                .strip_prefix(heads_root)
318                .unwrap_or(&path)
319                .to_string_lossy()
320                .replace('\\', "/");
321            let commit_id = read_ref(git_abs, &path)
322                .with_context(|| format!("读取分支 ref {}", path.display()))?
323                .commit_id;
324            out.push((name, commit_id));
325        }
326    }
327    Ok(())
328}

330/// HEAD 跟随分支时返回分支名(如 `master`);detached 时返回 `None`。
331fn current_branch_name(head: &Head) -> Option<String> {
332    match head {
333        Head::TargetBranch(symref) => Some(branch_name_of(&symref.ref_path)),
334        Head::TargetCommit(_) => None,
335    }
336}

338/// 把 `refs/heads/master` 这样的 ref 路径取末段为分支名;嵌套则保留相对 `refs/heads/` 的部分。
339fn branch_name_of(ref_path: &Path) -> String {
340    let s = ref_path.to_string_lossy().replace('\\', "/");
341    s.strip_prefix("refs/heads/").unwrap_or(&s).to_string()
342}

344/// 按 committer 时间升序排序一组 oid hex(时间相同则按 hex),让分叉输出稳定且「先发生的在前」。
345fn sort_by_time(hexes: &mut [String], commits: &HashMap<String, CommitObject>) {
346    hexes.sort_by(|a, b| {
347        let ta = commits.get(a).map(|c| c.committer.unix_time).unwrap_or(0);
348        let tb = commits.get(b).map(|c| c.committer.unix_time).unwrap_or(0);
349        ta.cmp(&tb).then_with(|| a.cmp(b))
350    });
351}

353/// 从 commit message 正文提取 `(摘要, 可选的上下文过滤说明)`。
354///
355/// - 摘要 = 第一行非空内容(即 `ask` 写入的问题概述);
356/// - 若正文里有 `Giftai-context: rounds X` 这条 trailer,转成 `仅基于第 X 轮` 的提示。
357fn commit_summary(message: &[u8]) -> (String, Option<String>) {
358    let text = String::from_utf8_lossy(message);
359    let summary = text
360        .lines()
361        .map(str::trim)
362        .find(|l| !l.is_empty())
363        .unwrap_or("(无摘要)")
364        .to_string();
365
366    let ctx_note = text
367        .lines()
368        .find_map(|l| l.trim().strip_prefix(CONTEXT_TRAILER_PREFIX))
369        .map(|rounds| format!("仅基于第 {} 轮", rounds.trim()));
370
371    (summary, ctx_note)
372}

374/// 数某个 tree(快照)里的轮数 = `NNNN-user.txt` 文件个数(每轮恰好一个 user 文件)。
375fn snapshot_rounds(git_abs: &Path, tree_oid: &ObjectSha) -> Result<u32> {
376    let tree = TreeObject::read_loose_tree(git_abs, &tree_oid.to_string())?;
377    let n = tree
378        .entries()
379        .keys()
380        .filter(|k| k.to_string_lossy().ends_with("-user.txt"))
381        .count();
382    Ok(n as u32)
383}

386fn short_hex(oid: &ObjectSha) -> String {

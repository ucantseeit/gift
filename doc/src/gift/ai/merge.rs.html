1//! `giftai merge <分支>`:把另一条对话线合并进当前线。
2//!
3//! 与代码 merge 不同,giftai 的对话是**只追加、永不修改**的(每轮只新增 `NNNN` 文件),两条线
4//! 共享前缀逐字节相同、各自新增的轮次彼此独立,所以**永远不会有内容冲突**,用不到三方合并。
5//! merge 在这里纯粹是「把两条线在 merge base 之后的轮次,按时间交错、重新编号,拼成一条」。
6//!
7//! 算法(设 base = 两 tip 的 merge base,R = 轮数 = first-parent 链深度):
8//! 1. `base == theirs`:对方已是祖先 → 无事可做。
9//! 2. `base == ours` :本线是对方祖先 → **fast-forward**(移动分支 ref + 刷新工作区,不造合并节点)。
10//! 3. 否则真合并:取两条线 base 之后的轮次,合并后**按提交时间排序**(同刻 ours 在前),
11//!    依次重新编号为 `base_rounds+1, +2, …`;据此重写工作区,造一个**双 parent**(ours、theirs)
12//!    的合并节点,message 为 `Merge <theirs> into <ours>`。
13//!
14//! 轮号 = first-parent 链深度,隐含假设**被合并的两条分支自身是线性的**(尚无嵌套合并节点);
15//! 若检测到 base 不落在链的对应深度,会明确报「暂不支持」而非给出错误结果。
16

36/// 一条「base 之后的轮次」引用:来自哪条线、原轮号、提交时间、所属 tip(用于读它的快照内容)。
37struct RoundRef {
38    side: Side,
39    old_round: u32,
40    time: i64,
41    tip: ObjectSha,
42}
43
44#[derive(Clone, Copy, PartialEq)]
45enum Side {
46    Ours,
47    Theirs,
48}
49
50impl Side {
51    /// 同一时刻的稳定排序:ours 在 theirs 之前。
52    fn rank(self) -> u8 {
53        match self {
54            Side::Ours => 0,
55            Side::Theirs => 1,
56        }
57    }
58}

60/// 把另一分支 `their_branch` 合并进当前分支。详见模块文档。
61pub fn merge(worktree: &Path, git_abs: &Path, their_branch: &str) -> Result<()> {
62    // 1. 必须站在某个分支上(要并入「当前分支」),且不能自合并。
63    let head = Head::read(git_abs).context("读取 HEAD 失败")?;
64    let our_branch = match &head {
65        Head::TargetBranch(symref) => branch_name_of(&symref.ref_path),
66        Head::TargetCommit(_) => bail!("merge 需在某个分支上进行(当前为游离 HEAD)"),
67    };
68    if our_branch == their_branch {
69        bail!("不能把分支合并到它自己:{their_branch}");
70    }
71    let our_tip = head.current_commit(git_abs).context("当前分支还没有任何提交")?;
72
73    // 2. 解析对方分支 tip。
74    let their_ref_abs = git_abs.join(get_branch_ref_path(their_branch));
75    let their_tip = read_ref(git_abs, &their_ref_abs)
76        .with_context(|| format!("找不到分支 {their_branch}"))?
77        .commit_id;
78
79    // 3. 求 merge base。
80    let base = find_merge_base(git_abs, &our_tip, &their_tip)?
81        .context("两条线没有共同祖先,无法合并")?;
82
83    // 4. 平凡情形。
84    if base == their_tip {
85        println!("已包含 {their_branch},无需合并。");
86        return Ok(());
87    }
88    if base == our_tip {
89        // ours 是 theirs 的祖先 → fast-forward:移动当前分支 ref 到 theirs,再刷新工作区。
90        let our_ref_abs = git_abs.join(get_branch_ref_path(&our_branch));
91        write_ref(git_abs, &our_ref_abs, &their_tip)
92            .with_context(|| format!("快进 {our_branch} 失败"))?;
93        checkout(worktree, git_abs, CheckoutTarget::Branch(our_branch.clone()))
94            .context("快进后刷新工作区失败")?;
95        println!("Fast-forward:{our_branch} 直接前进到 {their_branch}(未造合并节点)。");
96        return Ok(());
97    }
98
99    // 5. 真合并。先取两条线的 first-parent 链(根在前),据此算 base 深度与「base 之后的轮次」。
100    let ours_chain = first_parent_chain(git_abs, &our_tip)?;
101    let theirs_chain = first_parent_chain(git_abs, &their_tip)?;
102    let base_rounds = first_parent_chain(git_abs, &base)?.len() as u32;
103
104    // 安全校验:base 必须正好落在两条链的第 base_rounds 个位置(即两线在此之前完全一致)。
105    // 否则说明分支里已有嵌套合并/非线性历史,本实现的「轮号 = 深度」假设不成立。
106    ensure_linear_at_base(&ours_chain, &theirs_chain, base_rounds, &base)?;
107
108    // base 之后的轮次:链上深度 > base_rounds 的节点,各自带提交时间。
109    let mut after: Vec<RoundRef> = Vec::new();
110    after.extend(rounds_after(&ours_chain, base_rounds, Side::Ours, &our_tip));
111    after.extend(rounds_after(&theirs_chain, base_rounds, Side::Theirs, &their_tip));
112
113    // 按时间交错排序(同刻 ours 在前,再按旧轮号),得到新的线性顺序。
114    after.sort_by(|a, b| {
115        a.time
116            .cmp(&b.time)
117            .then(a.side.rank().cmp(&b.side.rank()))
118            .then(a.old_round.cmp(&b.old_round))
119    });
120
121    // 6. 重写工作区:删掉本线 base 之后的旧轮文件,再按新轮号写入交错后的内容。
122    let ours_total = base_rounds + ours_chain[base_rounds as usize..].len() as u32;
123    remove_rounds(worktree, base_rounds + 1, ours_total)?;
124
125    println!("Merge {their_branch} into {our_branch}");
126    for (i, r) in after.iter().enumerate() {
127        let new_round = base_rounds + 1 + i as u32;
128        let (user, assistant) = read_round_files(git_abs, &r.tip, r.old_round)?;
129        let (user_seq, asst_seq) = round_seqs(new_round);
130        fs::write(worktree.join(message_filename(user_seq, Role::User)), &user)
131            .with_context(|| format!("写入轮{new_round} user"))?;
132        fs::write(worktree.join(message_filename(asst_seq, Role::Assistant)), &assistant)
133            .with_context(|| format!("写入轮{new_round} assistant"))?;
134
135        let origin = match r.side {
136            Side::Ours => our_branch.as_str(),
137            Side::Theirs => their_branch,
138        };
139        println!("  轮{new_round} ← {origin} 旧轮{}  {}", r.old_round, first_line(&user));
140    }
141
142    // 7. 重建 index(本仓库 staging 不处理删除,故先清空再整盘重 stage,确保 tree 不残留旧文件)。
143    let idx_path = git_abs.join("index");
144    index_file::write_index_file(&idx_path, &IndexFile::empty(2))
145        .with_context(|| format!("重置 index {}", idx_path.display()))?;
146    let resolved = resolve_stage_inputs(&[worktree.to_path_buf()], worktree, git_abs)?;
147    stage_paths(git_abs, worktree, &resolved, true).context("暂存合并后的工作区失败")?;
148
149    // 8. 写 MERGE_HEAD,让 commit() 把它当作第二 parent(提交后由 commit() 自行清理)。
150    fs::write(git_abs.join("MERGE_HEAD"), format!("{}\n", their_tip.to_string()))
151        .context("写 MERGE_HEAD 失败")?;
152
153    let (author, committer) = identities_from_git_env()?;
154    let message = format!("Merge {their_branch} into {our_branch}");
155    let new_oid = commit(worktree, git_abs, author, committer, Some(message))?;
156
157    println!("合并完成:{}", new_oid.to_string());
158    Ok(())
159}

163/// 一轮对应的两个文件序号:user = 2r,assistant = 2r+1(system 固定为 1)。
164fn round_seqs(round: u32) -> (u32, u32) {
165    (2 * round, 2 * round + 1)
166}
167
168/// 沿 first-parent 链从 `tip` 回溯到根,返回 `(oid, commit)` 列表,**根在前、tip 在后**;
169/// 故下标 `i` 处的节点轮号 = `i + 1`。
170fn first_parent_chain(git_abs: &Path, tip: &ObjectSha) -> Result<Vec<(ObjectSha, CommitObject)>> {
171    let mut chain = Vec::new();
172    let mut cur = tip.clone();
173    loop {
174        let commit = CommitObject::read_loose_commit(git_abs, &cur.to_string())
175            .with_context(|| format!("读取 commit {}", cur.to_string()))?;
176        let next = commit.parents.first().cloned();
177        chain.push((cur, commit));
178        match next {
179            Some(parent) => cur = parent,
180            None => break,
181        }
182    }
183    chain.reverse();
184    Ok(chain)
185}

187/// 校验两条链在第 `base_rounds` 个位置都正好是 `base`(即合并前两线完全一致)。
188fn ensure_linear_at_base(
189    ours: &[(ObjectSha, CommitObject)],
190    theirs: &[(ObjectSha, CommitObject)],
191    base_rounds: u32,
192    base: &ObjectSha,
193) -> Result<()> {
194    let idx = base_rounds as usize - 1;
195    let ok = ours.get(idx).map(|(o, _)| o == base).unwrap_or(false)
196        && theirs.get(idx).map(|(o, _)| o == base).unwrap_or(false);
197    if !ok {
198        bail!("暂不支持合并含嵌套合并/非线性历史的分支(merge base 不在 first-parent 链的预期深度)");
199    }
200    Ok(())
201}

203/// 从一条 root-first 链中取出 base 之后(深度 > `base_rounds`)的轮次。
204fn rounds_after(
205    chain: &[(ObjectSha, CommitObject)],
206    base_rounds: u32,
207    side: Side,
208    tip: &ObjectSha,
209) -> Vec<RoundRef> {
210    chain
211        .iter()
212        .enumerate()
213        .filter_map(|(i, (_, commit))| {
214            let round = (i + 1) as u32;
215            (round > base_rounds).then(|| RoundRef {
216                side,
217                old_round: round,
218                time: commit.committer.unix_time,
219                tip: tip.clone(),
220            })
221        })
222        .collect()
223}

225/// 删除工作区中轮号在 `[from, to]` 内的 user/assistant 文件(不存在时静默忽略)。
226fn remove_rounds(worktree: &Path, from: u32, to: u32) -> Result<()> {
227    for round in from..=to {
228        let (user_seq, asst_seq) = round_seqs(round);
229        let _ = fs::remove_file(worktree.join(message_filename(user_seq, Role::User)));
230        let _ = fs::remove_file(worktree.join(message_filename(asst_seq, Role::Assistant)));
231    }
232    Ok(())
233}

235/// 从 `tip` 的快照里读出第 `round` 轮的 (user, assistant) 两个文件内容。
236fn read_round_files(git_abs: &Path, tip: &ObjectSha, round: u32) -> Result<(Vec<u8>, Vec<u8>)> {
237    let commit = CommitObject::read_loose_commit(git_abs, &tip.to_string())?;
238    let tree = TreeObject::read_loose_tree(git_abs, &commit.tree.to_string())?;
239    let (user_seq, asst_seq) = round_seqs(round);
240    let user = lookup_blob(git_abs, &tree, &message_filename(user_seq, Role::User))?;
241    let assistant = lookup_blob(git_abs, &tree, &message_filename(asst_seq, Role::Assistant))?;
242    Ok((user, assistant))
243}

245/// 在(扁平的)tree 中按文件名找到 blob 并读出其内容。
246fn lookup_blob(git_abs: &Path, tree: &TreeObject, name: &str) -> Result<Vec<u8>> {
247    let key = OsString::from_vec(name.as_bytes().to_vec());
248    let entry = tree
249        .entries()
250        .get(&key)
251        .with_context(|| format!("快照里缺少文件 {name}"))?;
252    let hex = entry.object_name.to_string();
253    let mut reader = Object::open_loose_object_bufreader(git_abs, &hex)?;
254    BlobObject::read_blob_payload(&mut reader, &hex)
255}

257/// 取一段内容的首行非空文本(展示用)。
258fn first_line(content: &[u8]) -> String {
259    String::from_utf8_lossy(content)
260        .lines()
261        .map(str::trim)
262        .find(|l| !l.is_empty())
263        .unwrap_or("")
264        .to_string()
265}

267/// `refs/heads/master` → `master`(嵌套则保留相对 `refs/heads/` 的部分)。
268fn branch_name_of(ref_path: &Path) -> String {
269    let s = ref_path.to_string_lossy().replace('\\', "/");
270    s.strip_prefix("refs/heads/").unwrap_or(&s).to_string()

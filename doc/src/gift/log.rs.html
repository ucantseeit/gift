1//! 显示提交历史（类似 `git log`）
2
3use anyhow::{Context, Result};
4use std::path::Path;
5use crate::head::Head;
6use crate::object::{CommitObject, ObjectSha};
7/// # 参数
8///
9/// - `git_abs`: Git 目录的绝对路径（例如 `.git/` 目录的完整路径）。
10/// - `max_count`: 最多显示的提交数量。如果为 `None`，则显示所有可达的提交。
11/// 打印提交历史，从 HEAD 开始沿着 parent 链遍历。
12pub fn log(git_abs: &Path, max_count: Option<usize>) -> Result<()> {
13    let head = Head::read(git_abs).context("failed to read HEAD")?;
14    let mut oid = head.current_commit(git_abs).context("no commits yet")?;
15
16    let mut count = 0;
17    loop {
18        if let Some(limit) = max_count {
19            if count >= limit {
20                break;
21            }
22        }
23
24        let commit = CommitObject::read_loose_commit(git_abs, &oid.to_string())
25            .with_context(|| format!("read commit {}", hex::encode(oid.as_bytes())))?;
26
27        print_commit(&oid, &commit);
28        count += 1;
29
30        match commit.parents.first() {
31            Some(parent) => oid = parent.clone(),
32            None => break,
33        }
34    }
35
36    Ok(())
37}
38
39fn print_commit(oid: &ObjectSha, commit: &CommitObject) {
40    let hash = hex::encode(oid.as_bytes());
41    println!("commit {}", hash);
42    println!();
43
44    let message = extract_message(&commit.message);
45    for line in message.lines() {
46        println!("    {}", line);
47    }
48    println!();
49}
50
51fn extract_message(raw: &[u8]) -> String {
52    let s = String::from_utf8_lossy(raw);
53    let mut lines = s.lines();
54    let mut found_empty = false;
55    let mut msg_lines = Vec::new();
56
57    for line in lines.by_ref() {
58        if !found_empty {
59            if line.is_empty() {
60                found_empty = true;
61            }
62            continue;
63        }
64        msg_lines.push(line);
65    }
66
67    if msg_lines.is_empty() && !found_empty {
68        s.trim().to_string()
69    } else {
70        msg_lines.join("\n").trim().to_string()
71    }
72}

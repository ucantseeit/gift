5use anyhow::{Context, Ok, Result, bail};
6use std::{fs, path::{Path, PathBuf}};

8/// 将绝对路径转为相对 git_abs 的路径（如 `refs/heads/main`）
9pub fn abs_path_to_git_path(
10    worktree: &Path,
11    git_abs: &Path,
12    path: &Path,
13) -> Result<String> {
14    let abs = worktree.join(path);
15    let rel = abs.strip_prefix(&git_abs).with_context(|| {
16        format!(
17            "git path {} not under git dir {}",
18            path.display(),
19            git_abs.display()
20        )
21    })?;
22    let s = rel.to_string_lossy().replace('\\', "/");
23    let s = s.trim_start_matches('/').to_string();
24    if s.is_empty() {
25        bail!("empty ref name from {}", path.display());
26    }
27    Ok(s)
28}

30/// 输入: branch_name
31/// 输出: 其对应 ref 文件的相对 git 仓库的路径
32pub fn get_branch_ref_path(
33    branch_name: &str
34) -> PathBuf {
35    PathBuf::new()
36        .join("refs")
37        .join("heads")
38        .join(branch_name)
39}

41/// loose 对象文件路径：`…/objects/ab/cdef…`（`hex_oid` 为完整 hex，不含换行）
42pub fn loose_object_path(
43    git_abs: &Path, 
44    hex_oid: &str
45) -> PathBuf {
46    debug_assert!(
47        hex_oid.len() >= 3,
48        "hex_oid must be long enough for objects/xx/yy… layout"
49    );
50    git_abs
51        .join("objects")
52        .join(&hex_oid[0..2])
53        .join(&hex_oid[2..])
54}

56#[derive(Debug, Clone)]
57pub struct RepoPaths {
58    pub worktree: PathBuf,
59    pub git_abs: PathBuf,
60}

62///向上找到当前目录的.gift文件夹和worktree的相对路径
63pub fn discover_repo_from_cwd() -> Result<RepoPaths>{
64    //返回当前工作目录的绝对路径
65    let cwd = std::env::current_dir()?;
66    let mut cur = fs::canonicalize(&cwd)?;
67    loop{
68        let gift_path = cur.join(".gift");
69        if gift_path.is_dir(){
70            return Ok(RepoPaths{
71                worktree:cur,
72                git_abs: gift_path,
73            });
74        }
75        match cur.parent() {
76            Some(parent) => cur = parent.to_path_buf(),
77            None => break,
78        }
79    }
80    bail!("not a gift repository (or any parent): {}", cwd.display());
81}

83/// 向上搜索 giftai 对话仓库（`.giftai`），返回解耦布局下的 `RepoPaths`。
84///
85/// 与 [`discover_repo_from_cwd`] 的区别在于 worktree 的位置：
86/// - `git_abs` = 找到的 `.giftai` 目录本身（对话元数据：objects/ refs/ HEAD …）；
87/// - `worktree` = `.giftai` 的**兄弟目录** `<root>/chat`（仅存放消息文件）。
88///
89/// 这样 `.giftai` 放在项目根、对话内容隔离在 `chat/`，二者互不包含，
90/// `giftai` 命令可在项目根或任意子目录直接运行。
91///
92/// 注意：本函数只负责定位，不保证 `chat/` 已存在（由 `giftai init` 创建）。
93pub fn discover_chat_repo() -> Result<RepoPaths> {
94    let cwd = std::env::current_dir()?;
95    let mut cur = fs::canonicalize(&cwd)?;
96    loop {
97        let giftai_path = cur.join(".giftai");
98        if giftai_path.is_dir() {
99            let root = giftai_path
100                .parent()
101                .with_context(|| {
102                    format!("`.giftai` has no parent: {}", giftai_path.display())
103                })?
104                .to_path_buf();
105            return Ok(RepoPaths {
106                worktree: root.join("chat"),
107                git_abs: giftai_path,
108            });
109        }
110        match cur.parent() {
111            Some(parent) => cur = parent.to_path_buf(),
112            None => break,
113        }
114    }
115    bail!("not a giftai repository (or any parent): {}", cwd.display());
116}

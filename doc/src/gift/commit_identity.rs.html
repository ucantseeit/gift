1//! 可选的 author / committer 解析（例如从 Git 风格环境变量读取），供 CLI 等上层组合 config / env 后调用 [`crate::commit::commit`]。
2//!
3//! 与 Git 行为接近的规则：`GIT_COMMITTER_*` 均未设置时，committer 与 author 相同；否则分别读取（未设置的 name/email 回退到 author）。
4
5use std::time::{SystemTime, UNIX_EPOCH};
6
7use anyhow::{Context, Result, bail};
8use crate::object::CommitIdentity;
9
10/// 解析 `GIT_AUTHOR_DATE` / `GIT_COMMITTER_DATE`：`<unix 秒> <时区>`，例如 `1700000000 +0800`。
11fn parse_git_identity_date(s: &str) -> Result<(i64, String)> {
12    let s = s.trim();
13    let mut it = s.split_whitespace();
14    let unix: i64 = it
15        .next()
16        .context("date string empty")?
17        .parse()
18        .context("bad unix timestamp in date")?;
19    let tz = it
20        .next()
21        .context("date string missing timezone after unix time")?
22        .to_string();
23    if it.next().is_some() {
24        bail!("date string has extra fields (expected `<unix> <tz>`)");
25    }
26    Ok((unix, tz))
27}
28
29fn now_unix_and_tz() -> (i64, String) {
30    let unix = SystemTime::now()
31        .duration_since(UNIX_EPOCH)
32        .unwrap_or_default()
33        .as_secs() as i64;
34    (unix, "+0000".into())
35}

37fn author_from_git_env() -> Result<CommitIdentity> {
38    let name = std::env::var("GIT_AUTHOR_NAME").unwrap_or_else(|_| "gift".into());
39    let email = std::env::var("GIT_AUTHOR_EMAIL").unwrap_or_else(|_| "gift@localhost".into());
40    let (unix_time, tz) = match std::env::var("GIT_AUTHOR_DATE") {
41        Ok(s) => parse_git_identity_date(&s).context("GIT_AUTHOR_DATE")?,
42        Err(_) => now_unix_and_tz(),
43    };
44    Ok(CommitIdentity {
45        name,
46        email,
47        unix_time,
48        tz,
49    })
50}
51
52fn committer_from_git_env(author: &CommitIdentity) -> Result<CommitIdentity> {
53    let any_committer_env = std::env::var("GIT_COMMITTER_NAME").is_ok()
54        || std::env::var("GIT_COMMITTER_EMAIL").is_ok()
55        || std::env::var("GIT_COMMITTER_DATE").is_ok();
56    if !any_committer_env {
57        return Ok(author.clone());
58    }
59    let name = std::env::var("GIT_COMMITTER_NAME").unwrap_or_else(|_| author.name.clone());
60    let email = std::env::var("GIT_COMMITTER_EMAIL").unwrap_or_else(|_| author.email.clone());
61    let (unix_time, tz) = match std::env::var("GIT_COMMITTER_DATE") {
62        Ok(s) => parse_git_identity_date(&s).context("GIT_COMMITTER_DATE")?,
63        Err(_) => now_unix_and_tz(),
64    };
65    Ok(CommitIdentity {
66        name,
67        email,
68        unix_time,
69        tz,
70    })
71}

73/// 从当前进程环境变量构造 author / committer（`GIT_AUTHOR_*`、`GIT_COMMITTER_*`），与原先 `commit` 内建逻辑一致。
74pub fn identities_from_git_env() -> Result<(CommitIdentity, CommitIdentity)> {
75    let author = author_from_git_env().context("GIT_AUTHOR_*")?;
76    let committer = committer_from_git_env(&author).context("GIT_COMMITTER_*")?;
77    Ok((author, committer))
78}

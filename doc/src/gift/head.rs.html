1//! 仓库 `HEAD`：symbolic（跟分支）或 detached（直接 OID）。
2
3use anyhow::{Context, Result, bail};
4use std::fs;
5use std::io::Write;
6use std::path::Path;
7
8use crate::object::ObjectSha;
9use crate::reference::{read_ref, write_ref};
10use crate::symbolic_ref::{read_symbolic_ref, write_symbolic_ref, SymbolicRef};
11
12/// 与 `HEAD` 文件内容对应
13#[derive(Debug, Clone, PartialEq, Eq)]
14pub enum Head {
15    TargetBranch(SymbolicRef),
16    TargetCommit(ObjectSha),
17}
18
19impl Head {
20    /// 从磁盘读取并解析 `HEAD`
21    pub fn read(
22        git_abs: &Path
23    ) -> Result<Head> {
24        let head_path = git_abs.join("HEAD");
25        let content =
26            fs::read_to_string(&head_path).with_context(|| format!("read HEAD {}", head_path.display()))?;
27        let line = content.trim();
28        if line.starts_with("ref:") {
29            let symref = read_symbolic_ref(&head_path)?;
30            Ok(Head::TargetBranch(symref))
31        } else {
32            let line = line.trim_end_matches(['\r', '\n']);
33            if line.lines().nth(1).is_some() {
34                bail!("HEAD must be a single line");
35            }
36            if line.len() != 40 || !line.chars().all(|c| c.is_ascii_hexdigit()) {
37                bail!(
38                    "expected detached HEAD (40 hex) or symbolic ref at {}",
39                    head_path.display()
40                );
41            }
42            let bytes: [u8; 20] = hex::decode(line)
43                .with_context(|| format!("decode HEAD at {}", head_path.display()))?
44                .try_into()
45                .map_err(|v: Vec<u8>| anyhow::anyhow!("HEAD oid length {}", v.len()))?;
46            Ok(Head::TargetCommit(ObjectSha::SHA1(bytes)))
47        }
48    }
49
50    /// 当前检出的 commit（分支：读 `branch_ref_path`；detached：即 `HEAD` 内 OID）
51    pub fn current_commit(
52        &self, 
53        git_abs: &Path
54    ) -> Result<ObjectSha> {
55        match self {
56            Head::TargetBranch(symref) => {
57                Ok(
58                    read_ref(
59                        git_abs, &git_abs.join(&symref.ref_path)
60                    )?.commit_id
61                )
62            }
63            Head::TargetCommit(oid) => Ok(oid.clone()),
64        }
65    }
66
67    /// 将 `Head` 写回 `HEAD` 文件
68    pub fn write(
69        &self, 
70        worktree: &Path, 
71        git_abs: &Path
72    ) -> Result<()> {
73        let head_path = git_abs.join("HEAD");
74        match self {
75            Head::TargetBranch(symref) => {
76                write_symbolic_ref(worktree, &head_path, symref)?;
77            }
78            Head::TargetCommit(oid) => {
79                let full = worktree.join(&head_path);
80                if let Some(parent) = full.parent() {
81                    fs::create_dir_all(parent)
82                        .with_context(|| format!("mkdir {}", parent.display()))?;
83                }
84                let mut f =
85                    fs::File::create(&full).with_context(|| format!("write HEAD {}", full.display()))?;
86                write!(f, "{}\n", hex::encode(oid.as_bytes()))
87                    .with_context(|| format!("write HEAD {}", full.display()))?;
88            }
89        }
90        Ok(())
91    }
92
93    /// 新 commit 落盘后：分支只更新 tip ref；detached 重写 `HEAD` 内 OID
94    pub fn record_new_commit(
95        &self,
96        worktree: &Path,
97        git_abs: &Path,
98        new_oid: &ObjectSha,
99    ) -> Result<()> {
100        match self {
101            Head::TargetBranch(symref) => {
102                write_ref(git_abs, &git_abs.join(&symref.ref_path), new_oid)?;
103            }
104            Head::TargetCommit(_) => {
105                Head::TargetCommit(new_oid.clone()).write(worktree, git_abs)?;
106            }
107        }
108        Ok(())
109    }
110}

1//! 直接 OID 的 ref（`git update-ref`）；路径相对 **worktree 根**。
2
3use anyhow::{Context, Result, bail};
4use std::{fs};
5use std::io::Write;
6use std::path::Path;

8use crate::object::{Object, ObjectSha};

10/// 对齐文件在磁盘上的语义：一行 40 位 hex（commit OID）。路径由 `read_ref` / `update_ref` 的参数传入，不放在本结构里。
11#[derive(Debug, Clone, PartialEq, Eq)]
12pub struct Ref {
13    pub commit_id: ObjectSha,
14}
15
16/// 读取 ref
17/// `git_abs` 用于定位 `objects/` 做类型校验
18pub fn read_ref(
19    git_abs: &Path,
20    ref_abs: &Path,
21) -> Result<Ref> {
22    assert!(ref_abs.is_absolute() && git_abs.is_absolute());
23
24    let content = fs::read_to_string(&ref_abs)
25        .with_context(|| format!("read {}", ref_abs.display()))?;
26    
27    // 检验并提取 content 里包含的 commit_id
28    let line = content.trim();
29    if line.starts_with("ref:") {
30        bail!(
31            "expected direct ref (hex oid) at {}, found symbolic ref",
32            ref_abs.display()
33        );
34    }
35    if line.lines().nth(1).is_some() {
36        bail!("ref file must be a single line: {}", ref_abs.display());
37    }
38    if line.len() != 40 {
39        bail!(
40            "bad SHA1 ref length {} at {}",
41            line.len(),
42            ref_abs.display()
43        );
44    }
45    if !line.chars().all(|c| c.is_ascii_hexdigit()) {
46        bail!("non-hex ref content at {}", ref_abs.display());
47    }
48    let bytes: [u8; 20] = hex::decode(line)
49        .with_context(|| format!("decode ref at {}", ref_abs.display()))?
50        .try_into()
51        .map_err(|v: Vec<u8>| anyhow::anyhow!("oid length {}", v.len()))?;
52
53    let commit_id = ObjectSha::SHA1(bytes);
54    let mut reader = 
55        Object::open_loose_object_bufreader(&git_abs, &commit_id.to_string())?;
56    Object::ensure_loose_object_kind(
57        &mut reader,
58        "commit", 
59        "read_ref")?;
60
61    Ok(Ref { commit_id })
62}

64/// 把 commit_id 写入指定的 ref 文件中
65/// ref_rel 是相对于 git_abs 的路径
66/// git_abs 用于检验 commit_id 是否有对应的object
67pub fn write_ref(
68    git_abs: &Path,
69    ref_abs: &Path,
70    commit_id: &ObjectSha,
71) -> Result<Ref> {
72    assert!(git_abs.is_absolute() && ref_abs.is_absolute());
73    
74    // 检验 commit_id 是否有对应的object
75    let mut reader = 
76        Object::open_loose_object_bufreader(&git_abs, &commit_id.to_string())?;
77    Object::ensure_loose_object_kind(
78        &mut reader, 
79        "commit", 
80        "update_ref")?;
81
82    // 创建 ref_rel的祖先目录
83    if let Some(parent) = ref_abs.parent() {
84        fs::create_dir_all(parent)
85            .with_context(|| format!("mkdir {}", parent.display()))?;
86    }
87
88    // 创建 ref_abs这个文件
89    let mut f =
90        fs::File::create(&ref_abs)
91        .with_context(|| format!("create ref {}", ref_abs.display()))?;
92
93    // 写入 ref 内容 (commit_id)
94    let hex = hex::encode(commit_id.as_bytes());
95    write!(f, "{hex}\n").with_context(|| format!("write {}", ref_abs.display()))?;
96    Ok(Ref {
97        commit_id: commit_id.clone(),
98    })
99}

101pub fn branch(
102    git_abs: &Path,
103    head_abs: &Path, 
104    branch_name: &str
105) -> Result<()> {
106    fn parse_oid_line(raw: &str, path: &Path) -> Result<String> {
107        let line = raw.trim();
108        if line.lines().nth(1).is_some() {
109            bail!("ref must be a single line: {}", path.display());
110        }
111        if line.len() != 40 || !line.chars().all(|c| c.is_ascii_hexdigit()) {
112            bail!("ref must be 40 hex chars: {}", path.display());
113        }
114        Ok(line.to_string())
115    }
116
117    let name = branch_name.trim();
118    if name.is_empty() {
119        bail!("branch name is empty");
120    }
121
122    let new_ref_abs = git_abs.join("refs").join("heads").join(name);
123    if new_ref_abs.exists() {
124        bail!("branch already exists: {}", name);
125    }
126
127    // 1) 读 HEAD：可能是 symbolic（ref: ...）也可能是 detached（40hex）
128    let head_raw = fs::read_to_string(head_abs)
129        .with_context(|| format!("read HEAD {}", head_abs.display()))?;
130    let head_line = head_raw.trim();
131
132    let tip_oid = if let Some(target) = head_line.strip_prefix("ref:") {
133        let target = target.trim();
134        if target.is_empty() {
135            bail!("empty HEAD symbolic target: {}", head_abs.display());
136        }
137        let target_ref = git_abs.join(target);
138        let raw = fs::read_to_string(&target_ref).with_context(|| {
139            format!(
140                "read target ref {} (unborn HEAD?)",
141                target_ref.display()
142            )
143        })?;
144        parse_oid_line(&raw, &target_ref)?
145    } else {
146        parse_oid_line(head_line, head_abs)?
147    };
148
149    // 2) 校验 tip oid 确实是 commit
150    let mut reader = 
151        Object::open_loose_object_bufreader(git_abs, &tip_oid.to_string())?;
152    Object::ensure_loose_object_kind(&mut reader, "commit", "branch")?;
153
154    // 3) 创建新分支 ref，要求不存在（create_new 防覆盖）
155    if let Some(parent) = new_ref_abs.parent() {
156        fs::create_dir_all(parent).with_context(|| format!("mkdir {}", parent.display()))?;
157    }
158    let mut f = fs::OpenOptions::new()
159        .write(true)
160        .create_new(true)
161        .open(&new_ref_abs)
162        .with_context(|| format!("create branch ref {}", new_ref_abs.display()))?;
163    write!(f, "{tip_oid}\n").with_context(|| format!("write {}", new_ref_abs.display()))?;
164
165    Ok(())
166}

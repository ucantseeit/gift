1//! `git push`：把本地分支的新对象发给远端，并请求远端更新对应的 ref。
2//! 走的是 `git-receive-pack` 服务（与 fetch/clone 的 `git-upload-pack` 相反方向）。
3//!
4//! 流程（与 fetch 镜像）：
5//!   1. discover：GET `…/info/refs?service=git-receive-pack`，拿到远端各 ref 的**当前值**；
6//!   2. 算出本次要更新的 ref：`old`（远端当前值，新分支则全 0）→ `new`（本地分支 tip）；
7//!   3. 快进检查（非 force 时）：`old` 不是 `new` 的祖先就拒绝，提示先 pull；
8//!   4. 遍历对象图，算出「`new` 可达但远端已有的对象之外」的对象集合；
9//!   5. 用 [`build_pack`] 把它们打成 packfile；
10//!   6. POST `…/git-receive-pack`：命令行 `<old> <new> <ref>` + flush + packfile；
11//!   7. 解析 report-status，成功则把本地 `refs/remotes/<remote>/<branch>` 也推进到 `new`。
12//!
13//! 复用：[`discover_refs_for`]（discovery）、[`build_pack`] / [`read_loose_object_by_hex`]
14//! （打包 / 读对象）、[`find_merge_base`]（快进判定）、pkt-line 编解码、[`current_branch_name`]。
15//!
16//! 简化：假设对象都是松散对象（与本项目其它写路径一致，从不生成 pack）；
17//! 子模块 gitlink（mode 160000）指向的 commit 不在本库，遍历时跳过。

34/// 单个 ref 的推送结果（用于打印摘要）。

38    pub new: String,

42/// 该 hex 对应的松散对象是否存在于本地对象库。
43fn object_exists(git_abs: &Path, hex: &str) -> bool {
44    hex.len() >= 3 && git_abs.join("objects").join(&hex[0..2]).join(&hex[2..]).exists()
45}

47fn sha_from_hex(hex: &str) -> Result<ObjectSha> {
48    let bytes: [u8; 20] = hex::decode(hex)
49        .context("OID hex decode")?
50        .try_into()
51        .map_err(|_| anyhow::anyhow!("OID must be 20 bytes"))?;
52    Ok(ObjectSha::SHA1(bytes))
53}

55/// 读 `refs/heads/<branch>` 的 OID。
56fn local_branch_oid(git_abs: &Path, branch: &str) -> Result<String> {
57    let p = git_abs.join("refs").join("heads").join(branch);
58    let s = fs::read_to_string(&p)
59        .with_context(|| format!("本地分支不存在：{}", p.display()))?;
60    Ok(s.trim().to_string())
61}

63/// 从 commit payload 取 `tree` 与 `parent` 的 hex。
64fn commit_children(payload: &[u8]) -> (Option<String>, Vec<String>) {
65    let text = String::from_utf8_lossy(payload);
66    let (mut tree, mut parents) = (None, Vec::new());
67    for line in text.lines() {
68        if line.is_empty() {
69            break; // header 结束
70        }
71        if let Some(h) = line.strip_prefix("tree ") {
72            tree = Some(h.trim().to_string());
73        } else if let Some(h) = line.strip_prefix("parent ") {
74            parents.push(h.trim().to_string());
75        }
76    }
77    (tree, parents)
78}

80/// 从 tree payload 取每个条目的 `(mode, child hex)`。
81fn tree_children(payload: &[u8]) -> Vec<(u32, String)> {
82    let mut out = Vec::new();
83    let mut pos = 0;
84    while pos < payload.len() {
85        let sp = match payload[pos..].iter().position(|&b| b == b' ') {
86            Some(i) => pos + i,
87            None => break,
88        };
89        let mode = std::str::from_utf8(&payload[pos..sp])
90            .ok()
91            .and_then(|s| u32::from_str_radix(s, 8).ok())
92            .unwrap_or(0);
93        let nul = match payload[sp + 1..].iter().position(|&b| b == 0) {
94            Some(i) => sp + 1 + i,
95            None => break,
96        };
97        if nul + 21 > payload.len() {
98            break;
99        }
100        let hex: String = payload[nul + 1..nul + 21].iter().map(|b| format!("{b:02x}")).collect();
101        out.push((mode, hex));
102        pos = nul + 21;
103    }
104    out
105}

107/// 把从 `root` 可达的对象 OID 全部标进 `present`（迭代式，避免深递归）。
108/// 用于标记“远端已有”的边界：root 取远端广告里、且本地也有的 ref。
109fn mark_present(git_abs: &Path, root: &str, present: &mut HashSet<String>) -> Result<()> {
110    let mut stack = vec![root.to_string()];
111    while let Some(oid) = stack.pop() {
112        if oid == ZERO_OID || !object_exists(git_abs, &oid) || !present.insert(oid.clone()) {
113            continue;
114        }
115        let (kind, payload) = match read_loose_object_by_hex(git_abs, &oid) {
116            Ok(x) => x,
117            Err(_) => continue, // 本地没有就当作边界，不展开
118        };
119        push_children(kind, &payload, &mut stack);
120    }
121    Ok(())
122}
123
124/// 从 `new` 出发收集要发送的对象：凡不在 `present`（远端已有）里的都要发。
125/// 遇到 present 里的对象直接剪枝（其整棵可达子图远端都有）。
126fn collect_to_send(
127    git_abs: &Path,
128    new: &str,
129    present: &HashSet<String>,
130) -> Result<Vec<(Kind, Vec<u8>)>> {
131    let mut queued: HashSet<String> = HashSet::new();
132    let mut out = Vec::new();
133    let mut stack = vec![new.to_string()];
134    while let Some(oid) = stack.pop() {
135        if oid == ZERO_OID || present.contains(&oid) || !queued.insert(oid.clone()) {
136            continue;
137        }
138        let (kind, payload) = read_loose_object_by_hex(git_abs, &oid)?;
139        push_children(kind, &payload, &mut stack);
140        out.push((kind, payload));
141    }
142    Ok(out)
143}
144
145/// 把一个对象的子对象 OID 压栈（commit→tree+parents；tree→非 gitlink 条目）。
146fn push_children(kind: Kind, payload: &[u8], stack: &mut Vec<String>) {
147    match kind {
148        Kind::Commit => {
149            let (tree, parents) = commit_children(payload);
150            if let Some(t) = tree {
151                stack.push(t);
152            }
153            stack.extend(parents);
154        }
155        Kind::Tree => {
156            for (mode, child) in tree_children(payload) {
157                if mode == 0o160000 {
158                    continue; // gitlink 子模块 commit，不在本库
159                }
160                stack.push(child);
161            }
162        }
163        Kind::Blob | Kind::Tag => {}
164    }
165}

167/// POST 命令 + packfile 到 `…/git-receive-pack`，整段读回响应。
168fn send_receive_pack(base: &str, body: &[u8]) -> Result<Vec<u8>> {
169    let url = format!("{}/git-receive-pack", base.trim_end_matches('/'));
170    let mut res = ureq::post(&url)
171        .header("Content-Type", "application/x-git-receive-pack-request")
172        .header("Accept", "application/x-git-receive-pack-result")
173        .send(body)?;
174    let bytes = res.body_mut().with_config().limit(64 * 1024 * 1024).read_to_vec()?;
175    Ok(bytes)
176}

178/// 解析 report-status：返回 `(unpack 是否 ok, 每个 ref 的 (refname, Option<拒绝原因>))`。
179fn parse_report_status(bytes: &[u8]) -> Result<(bool, Vec<(String, Option<String>)>)> {
180    let mut cur: &[u8] = bytes;
181    let mut unpack_ok = false;
182    let mut refs = Vec::new();
183    while let Some(line) = read_pkt_line(&mut cur)? {
184        let payload = match line {
185            PktLine::Data(p) => p,
186            _ => continue, // flush / delim
187        };
188        let s = String::from_utf8_lossy(&payload);
189        let s = s.trim_end();
190        if let Some(rest) = s.strip_prefix("unpack ") {
191            unpack_ok = rest == "ok";
192        } else if let Some(rest) = s.strip_prefix("ok ") {
193            refs.push((rest.to_string(), None));
194        } else if let Some(rest) = s.strip_prefix("ng ") {
195            let mut it = rest.splitn(2, ' ');
196            let name = it.next().unwrap_or("").to_string();
197            let reason = it.next().unwrap_or("(no reason)").to_string();
198            refs.push((name, Some(reason)));
199        }
200    }
201    Ok((unpack_ok, refs))
202}

204/// 顶层 push：把本地 `branch`（默认当前分支）推到远端同名分支。
205pub fn push(
206    worktree: &Path,
207    git_abs: &Path,
208    url: &str,
209    remote: &str,
210    branch: Option<&str>,
211    force: bool,
212) -> Result<()> {
213    let _ = worktree;
214    let branch = match branch {
215        Some(b) => b.to_string(),
216        None => current_branch_name(git_abs)?,
217    };
218    let new = local_branch_oid(git_abs, &branch)?;
219    let remote_refname = format!("refs/heads/{branch}");
220
221    // 1. discover：远端这个 ref 的当前值（没有则为全 0 = 新建分支）
222    let adv = discover_refs_for(url, "git-receive-pack")?;
223    let old = adv.sha_of(&remote_refname).unwrap_or(ZERO_OID).to_string();
224
225    if old == new {
226        println!("Everything up-to-date");
227        return Ok(());
228    }
229
230    // 2. 快进检查：仅当本地有 old 对象、且未 --force 时做（否则交给远端裁决）
231    if !force && old != ZERO_OID && object_exists(git_abs, &old) {
232        let base = find_merge_base(git_abs, &sha_from_hex(&old)?, &sha_from_hex(&new)?)?;
233        let is_ff = base.map(|b| b.to_string()).as_deref() == Some(old.as_str());
234        if !is_ff {
235            bail!(
236                "更新被拒绝：非快进推送（远端 {} 含有你本地没有的提交）。\n\
237                 先 `gift pull` 合并，或用 `--force` 强推。",
238                &old[..old.len().min(7)]
239            );
240        }
241    }
242
243    // 3. 标记“远端已有”的边界：广告里所有、且本地也存在的 ref 都展开
244    let mut present = HashSet::new();
245    for r in &adv.refs {
246        if object_exists(git_abs, &r.sha) {
247            mark_present(git_abs, &r.sha, &mut present)?;
248        }
249    }
250
251    // 4. 收集要发的对象 → 打包
252    let objects = collect_to_send(git_abs, &new, &present)?;
253    let pack = build_pack(&objects)?;
254
255    // 5. 拼请求体：命令行（首行带能力位）+ flush + packfile
256    let mut body = Vec::new();
257    let cmd = format!("{old} {new} {remote_refname}\0report-status\n");
258    write_pkt_line(&mut body, &PktLine::Data(cmd.into_bytes()))?;
259    write_pkt_line(&mut body, &PktLine::Flush)?;
260    body.extend_from_slice(&pack);
261
262    // 6. 发送 + 解析 report-status
263    let resp = send_receive_pack(url, &body)?;
264    let (unpack_ok, ref_status) = parse_report_status(&resp)?;
265    if !unpack_ok {
266        bail!("远端 unpack 失败（pack 校验未通过）");
267    }
268    let rejected = ref_status
269        .iter()
270        .find(|(rn, _)| rn == &remote_refname)
271        .and_then(|(_, reason)| reason.clone());
272
273    // 7. 推送成功则把本地远程跟踪引用也推进到 new
274    if rejected.is_none() {
275        let track = git_abs.join("refs").join("remotes").join(remote).join(&branch);
276        if let Some(p) = track.parent() {
277            fs::create_dir_all(p)?;
278        }
279        fs::write(&track, format!("{new}\n"))?;
280    }
281
282    // 打印摘要
283    let result = PushResult { refname: remote_refname, old: old.clone(), new: new.clone(), rejected };
284    print_summary(url, &branch, &result);
285    if result.rejected.is_some() {
286        std::process::exit(1);
287    }
288    Ok(())
289}
290
291fn short(h: &str) -> &str {
292    &h[..h.len().min(7)]
293}
294
295fn print_summary(url: &str, branch: &str, r: &PushResult) {
296    println!("To {url}");
297    match &r.rejected {
298        Some(reason) => {
299            println!(" ! [rejected]        {branch} -> {branch} ({reason})");
300        }
301        None if r.old == ZERO_OID => {
302            println!(" * [new branch]      {branch} -> {branch}");
303        }
304        None => {
305            println!("   {}..{}  {branch} -> {branch}", short(&r.old), short(&r.new));
306        }
307    }
308}

324    /// 离线验证 push 的两个核心：①对象图差分（collect_to_send）算出的集合，应与
325    /// git 的权威答案 `git rev-list --objects B --not A` 完全一致；②build_pack 造的
326    /// 包能被 parse_pack 解回（顺带验 OID）。两者串起来就是“要发的 pack 内容正确”。
327    #[test]
328    fn objects_to_send_match_git_rev_list() {
329        use crate::parse_packfile::parse_pack;
330
331        let env = [
332            ("GIT_AUTHOR_NAME", "t"), ("GIT_AUTHOR_EMAIL", "t@t"),
333            ("GIT_COMMITTER_NAME", "t"), ("GIT_COMMITTER_EMAIL", "t@t"),
334        ];
335        let src = std::env::temp_dir().join(format!("push_{}", std::process::id()));
336        let _ = fs::remove_dir_all(&src);
337        run(Command::new("git").args(["init", "-q"]).arg(&src));
338
339        // 提交 A：根文件 + 子目录文件
340        fs::create_dir_all(src.join("sub")).unwrap();
341        fs::write(src.join("root.txt"), "root v1\n").unwrap();
342        fs::write(src.join("sub/a.txt"), "a v1\n").unwrap();
343        run(Command::new("git").arg("-C").arg(&src).args(["add", "."]));
344        run(Command::new("git").arg("-C").arg(&src).args(["commit", "-q", "-m", "A"]).envs(env));
345        let a = git_out(&src, &["rev-parse", "HEAD"]).trim().to_string();
346
347        // 提交 B：只改子目录里的文件（root.txt 与其 blob 应保持不变 → 不该被重发）
348        fs::write(src.join("sub/a.txt"), "a v2 changed\n").unwrap();
349        run(Command::new("git").arg("-C").arg(&src).args(["add", "."]));
350        run(Command::new("git").arg("-C").arg(&src).args(["commit", "-q", "-m", "B"]).envs(env));
351        let b = git_out(&src, &["rev-parse", "HEAD"]).trim().to_string();
352
353        let git_abs = src.join(".git");
354
355        // 我方：present = 从 A 可达的一切；要发 = 从 B 可达且不在 present 的
356        let mut present = HashSet::new();
357        mark_present(&git_abs, &a, &mut present).unwrap();
358        let objs = collect_to_send(&git_abs, &b, &present).unwrap();
359
360        // 串 build_pack → parse_pack，拿到我方对象 OID 集合
361        let pack = build_pack(&objs).unwrap();
362        let ours: HashSet<String> =
363            parse_pack(&pack).unwrap().iter().map(|o| o.hex()).collect();
364
365        // git 权威答案：B 相对 A 新增的对象
366        let listed = git_out(&src, &["rev-list", "--objects", &b, "--not", &a]);
367        let expected: HashSet<String> = listed
368            .lines()
369            .filter_map(|l| l.split(' ').next())
370            .filter(|s| s.len() == 40)
371            .map(|s| s.to_string())
372            .collect();
373
374        assert_eq!(ours, expected, "要发送的对象集合应与 git rev-list 完全一致");
375        // 具体性检查：root.txt 的 blob（未改）不应在发送集合里
376        let root_blob = git_out(&src, &["rev-parse", &format!("{a}:root.txt")]).trim().to_string();
377        assert!(!ours.contains(&root_blob), "未改动的 root.txt blob 不应被重发");
378        // B 这个新 commit 必须在
379        assert!(ours.contains(&b), "新提交 B 必须被发送");
380
381        let _ = fs::remove_dir_all(&src);
382    }

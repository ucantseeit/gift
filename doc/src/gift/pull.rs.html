1//! `git pull` = `fetch` + `merge`。完全复用已有两块：
2//!   - [`crate::fetch::fetch`]：拉对象 + 更新 `refs/remotes/<remote>/*`（不动工作区）
3//!   - [`crate::merge::merge`]：把抓到的远程分支并入当前分支（FF / 三方 / 冲突）
4//!
5//! 简化点：本项目没有 remote / upstream 配置系统，所以远端 `url` 显式给出；
6//! 要并入的远程分支默认取“当前 HEAD 所在分支的同名远程分支”，也可显式指定。
7//!
8//! 拆成两层是为了可测：`pull` = `fetch` + `merge_remote_tracking`，后者（fetch 之后
9//! 的纯本地“并入”逻辑）不依赖网络，可离线测试。
10

21/// 取当前 HEAD 所跟随的本地分支短名（如 `"master"`）。detached HEAD 时报错。
22pub(crate) fn current_branch_name(git_abs: &Path) -> Result<String> {
23    match Head::read(git_abs)? {
24        Head::TargetBranch(symref) => {
25            let p = symref.ref_path.to_string_lossy().replace('\\', "/"); // refs/heads/master
26            Ok(p.strip_prefix("refs/heads/").unwrap_or(&p).to_string())
27        }
28        Head::TargetCommit(_) => {
29            bail!("当前处于 detached HEAD，请用 -b/--branch 显式指定要 pull 的分支")
30        }
31    }
32}

34/// fetch 之后的“并入”部分（纯本地）：
35/// 解析要并入的远程分支 → 读 `refs/remotes/<remote>/<branch>` 的 OID → 调 `merge`。
36///
37/// `branch` 为 `None` 时取当前分支同名。target 传 40 位 hex OID，`merge` 直接解析，
38/// 无需把远程分支写进 `refs/heads/`。
39pub fn merge_remote_tracking(
40    worktree: &Path,
41    git_abs: &Path,
42    remote: &str,
43    branch: Option<&str>,
44    author: CommitIdentity,
45    committer: CommitIdentity,
46    message: Option<String>,
47) -> Result<MergeOutcome> {
48    let branch = match branch {
49        Some(b) => b.to_string(),
50        None => current_branch_name(git_abs)?,
51    };
52
53    let track = git_abs.join("refs").join("remotes").join(remote).join(&branch);
54    let oid_hex = fs::read_to_string(&track)
55        .with_context(|| {
56            format!(
57                "远程跟踪分支不存在：{}（远端有 {branch} 这个分支、且已 fetch 过吗？）",
58                track.display()
59            )
60        })?
61        .trim()
62        .to_string();
63
64    let msg = message.unwrap_or_else(|| format!("Merge branch '{remote}/{branch}'\n"));
65    merge(worktree, git_abs, &oid_hex, author, committer, &msg)
66}

68/// 顶层 pull：先 `fetch`（拉对象 + 更新远程跟踪引用，并打印 fetch 摘要），
69/// 再把对应的远程分支并入当前分支。
70pub fn pull(
71    worktree: &Path,
72    git_abs: &Path,
73    url: &str,
74    remote: &str,
75    branch: Option<&str>,
76    author: CommitIdentity,
77    committer: CommitIdentity,
78    message: Option<String>,
79) -> Result<MergeOutcome> {
80    fetch(git_abs, url, remote)?; // 复用 fetch：拉对象 + 更新 refs/remotes/<remote>/*
81    merge_remote_tracking(worktree, git_abs, remote, branch, author, committer, message)
82}

89    fn run(c: &mut Command) {
90        assert!(c.status().unwrap().success(), "命令失败: {c:?}");

92    fn git_out(repo: &Path, args: &[&str]) -> String {
93        let out = Command::new("git").arg("-C").arg(repo).args(args).output().unwrap().stdout;
94        String::from_utf8(out).unwrap().trim().to_string()
95    }
96    fn ident() -> CommitIdentity {
97        CommitIdentity { name: "t".into(), email: "t@t".into(), unix_time: 0, tz: "+0000".into() }
98    }

100    /// 离线验证 fetch 之后的“并入”：本地 master 落后于 origin/master 时应快进，
101    /// 并真正更新本地分支 ref 与工作区文件。
102    #[test]
103    fn merge_remote_tracking_fast_forwards() {
104        let env = [
105            ("GIT_AUTHOR_NAME", "t"), ("GIT_AUTHOR_EMAIL", "t@t"),
106            ("GIT_COMMITTER_NAME", "t"), ("GIT_COMMITTER_EMAIL", "t@t"),
107        ];
108
109        let repo = std::env::temp_dir().join(format!("pull_{}", std::process::id()));
110        let _ = fs::remove_dir_all(&repo);
111        run(Command::new("git").args(["init", "-q"]).arg(&repo));
112
113        // 提交 A（旧），再提交 B（新）
114        fs::write(repo.join("data.txt"), "v1\n").unwrap();
115        run(Command::new("git").arg("-C").arg(&repo).args(["add", "."]));
116        run(Command::new("git").arg("-C").arg(&repo).args(["commit", "-q", "-m", "A"]).envs(env));
117        let a = git_out(&repo, &["rev-parse", "HEAD"]);
118        fs::write(repo.join("data.txt"), "v2\n").unwrap();
119        run(Command::new("git").arg("-C").arg(&repo).args(["add", "."]));
120        run(Command::new("git").arg("-C").arg(&repo).args(["commit", "-q", "-m", "B"]).envs(env));
121        let b = git_out(&repo, &["rev-parse", "HEAD"]);
122        let branch = git_out(&repo, &["rev-parse", "--abbrev-ref", "HEAD"]); // master / main
123
124        // 把本地分支退回 A（模拟“本地落后”），B 的对象仍在库中
125        run(Command::new("git").arg("-C").arg(&repo).args(["reset", "-q", "--hard", &a]));
126        assert_eq!(fs::read_to_string(repo.join("data.txt")).unwrap(), "v1\n");
127
128        // 模拟 fetch 的产物：远程跟踪引用 origin/<branch> 指向 B
129        let git_abs = repo.join(".git");
130        let track = git_abs.join("refs/remotes/origin").join(&branch);
131        fs::create_dir_all(track.parent().unwrap()).unwrap();
132        fs::write(&track, format!("{b}\n")).unwrap();
133
134        // “并入”部分：branch=None → 取当前分支 → 读 origin/<branch>=B → merge（应快进）
135        let outcome = merge_remote_tracking(
136            &repo, &git_abs, "origin", None, ident(), ident(), None,
137        ).unwrap();
138
139        match outcome {
140            MergeOutcome::FastForward(oid) => assert_eq!(oid.to_string(), b, "应快进到 B"),
141            MergeOutcome::AlreadyUpToDate => panic!("期望 FastForward，得到 AlreadyUpToDate"),
142            MergeOutcome::Clean(_) => panic!("期望 FastForward，得到 Clean"),
143            MergeOutcome::Conflict(_) => panic!("期望 FastForward，得到 Conflict"),
144        }
145        // 本地分支 ref 已前进到 B；工作区文件已更新为 v2
146        assert_eq!(git_out(&repo, &["rev-parse", "HEAD"]), b, "本地分支应指向 B");
147        assert_eq!(fs::read_to_string(repo.join("data.txt")).unwrap(), "v2\n", "工作区应更新为 v2");
148
149        let _ = fs::remove_dir_all(&repo);
150    }

152    /// 已经是最新时应报告 Already up to date（target 即当前 commit）。
153    #[test]
154    fn merge_remote_tracking_already_up_to_date() {
155        let env = [
156            ("GIT_AUTHOR_NAME", "t"), ("GIT_AUTHOR_EMAIL", "t@t"),
157            ("GIT_COMMITTER_NAME", "t"), ("GIT_COMMITTER_EMAIL", "t@t"),
158        ];
159        let repo = std::env::temp_dir().join(format!("pull2_{}", std::process::id()));
160        let _ = fs::remove_dir_all(&repo);
161        run(Command::new("git").args(["init", "-q"]).arg(&repo));
162        fs::write(repo.join("a"), "x\n").unwrap();
163        run(Command::new("git").arg("-C").arg(&repo).args(["add", "."]));
164        run(Command::new("git").arg("-C").arg(&repo).args(["commit", "-q", "-m", "A"]).envs(env));
165        let a = git_out(&repo, &["rev-parse", "HEAD"]);
166        let branch = git_out(&repo, &["rev-parse", "--abbrev-ref", "HEAD"]);
167
168        let git_abs = repo.join(".git");
169        let track = git_abs.join("refs/remotes/origin").join(&branch);
170        fs::create_dir_all(track.parent().unwrap()).unwrap();
171        fs::write(&track, format!("{a}\n")).unwrap(); // origin 与本地同一 commit
172
173        let outcome = merge_remote_tracking(
174            &repo, &git_abs, "origin", None, ident(), ident(), None,
175        ).unwrap();
176        assert!(matches!(outcome, MergeOutcome::AlreadyUpToDate));
177
178        let _ = fs::remove_dir_all(&repo);
179    }

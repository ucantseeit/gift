1//! # diff 模块
2//!
3//! ## 当前设计范围与取舍
4//!
5//! 本模块目前只处理**两侧都已在 object 数据库中**的场景：
6//! - `diff_trees` / `diff_dirs`：两棵 `TreeObject` 均来自 `.git/objects`
7//! - `diff_commits`：两个 commit 的 tree 均来自 `.git/objects`
8//! - `Added`/`Deleted` 的对端以 `BlobSide::Empty`（空内容）表示
9//!
10//! 树级结构 diff（`diff_trees`）只比较 OID 和 FileMode，**不读取 blob 内容**；
11//! 内容级 diff（`diff_dirs` 及以上）才通过 `BlobSide` 读取字节并调用 diffy。
12//!
13//! ## BlobSide：blob 内容来源抽象
14//!
15//! `BlobSide` 是唯一需要关心"从哪里读字节"的地方。目前有三个变体：
16//! - `Empty`：空内容（用于新增/删除文件的对端）
17//! - `Stored`：从 object 数据库读（当前唯一真正使用的场景）
18//! - `Worktree`：直接从磁盘文件读（**暂未在上层函数中使用**，为 `git diff`
19//!   工作区 vs 暂存区/commit 预留；实现时在 `diff_dirs` 层引入对应的
20//!   `TreeSide::Worktree` 变体，枚举磁盘目录条目即可）
21//!
22//! ## 未来扩展方向
23//!
24//! 1. **工作区 diff**（`git diff` 无参数）：
25//!    需引入 `TreeSide { Stored(&TreeObject), Worktree(&Path) }`，
26//!    `Worktree` 变体扫描磁盘目录、按需读文件字节（不需预写 object db）。
27//!    blob 内容读取直接用 `BlobSide::Worktree`，已有路径。
28//!
29//! 2. **暂存区 vs commit**（`git diff --cached`）：
30//!    index 中 blob OID 已写入 object db（`git add` 时写入），
31//!    可直接复用 `BlobSide::Stored`；只需把 index 转成类 `TreeObject`
32//!    的遍历接口即可接入现有 diff 栈。
33//!
34//! 3. **重命名检测**：在 `diff_trees` 返回的 `Deleted`+`Added` 对里
35//!    做相似度匹配（blob 内容 edit-distance），目前未实现。
36

45// ── BlobSide：blob 内容来源 ───────────────────────────────────────────────────

47/// blob 内容的来源，是 diff 模块唯一需要关心"从哪里读字节"的地方。
48pub enum BlobSide<'a> {
49    /// 空内容（新增文件的 old 侧 / 删除文件的 new 侧）
50    Empty,
51    /// 从 `.git/objects` 读取已存储的 blob
52    Stored(&'a ObjectSha),
53    /// 直接从磁盘读取工作区文件（为 `git diff` 工作区场景预留，暂未在上层使用）
54    Worktree(&'a Path),
55}
56
57fn read_blob_bytes(git_abs: &Path, side: &BlobSide<'_>) -> Result<Vec<u8>> {
58    match side {
59        BlobSide::Empty => Ok(Vec::new()),
60        BlobSide::Stored(oid) => {
61            let hex = oid.to_string();
62            let mut br = Object::open_loose_object_bufreader(git_abs, &hex)?;
63            BlobObject::read_blob_payload(&mut br, &hex)
64        }
65        BlobSide::Worktree(path) => {
66            fs::read(path).with_context(|| format!("read worktree file {}", path.display()))
67        }
68    }
69}

71// ── BlobDiff ─────────────────────────────────────────────────────────────────
72
73pub enum BlobDiff {
74    /// 可解析为 UTF-8 的文本 diff，内容为 unified diff 格式字符串
75    Text(String),
76    /// 至少一侧含 `\0` 字节（git 二进制检测规范），不做行级对比
77    Binary,
78}
79
80/// 比较两个 blob 来源的内容，返回 unified diff 字符串或二进制标记。
81pub fn diff_blobs(git_abs: &Path, old: BlobSide<'_>, new: BlobSide<'_>) -> Result<BlobDiff> {
82    let old_bytes = read_blob_bytes(git_abs, &old)?;
83    let new_bytes = read_blob_bytes(git_abs, &new)?;
84
85    if old_bytes.contains(&0u8) || new_bytes.contains(&0u8) {
86        return Ok(BlobDiff::Binary);
87    }
88    let old_str = match std::str::from_utf8(&old_bytes) {
89        Ok(s) => s,
90        Err(_) => return Ok(BlobDiff::Binary),
91    };
92    let new_str = match std::str::from_utf8(&new_bytes) {
93        Ok(s) => s,
94        Err(_) => return Ok(BlobDiff::Binary),
95    };
96
97    let patch = diffy::create_patch(old_str, new_str);
98    Ok(BlobDiff::Text(format!("{patch}")))
99}

101// ── DiffEntry：树级结构差异 ───────────────────────────────────────────────────

103/// 两棵 tree 之间单条路径的结构差异（只含 OID / FileMode，不含内容）。
104/// 路径相对于传入的两棵 tree 的公共根。
105#[derive(Debug, Clone)]
106pub enum DiffEntry {
107    Added   { path: PathBuf, mode: FileMode, oid: ObjectSha },
108    Deleted { path: PathBuf, mode: FileMode, oid: ObjectSha },
109    Modified {
110        path: PathBuf,
111        old_mode: FileMode,
112        new_mode: FileMode,
113        old_oid:  ObjectSha,
114        new_oid:  ObjectSha,
115    },
116}
117
118impl DiffEntry {
119    pub fn path(&self) -> &Path {
120        match self {
121            DiffEntry::Added   { path, .. } => path,
122            DiffEntry::Deleted { path, .. } => path,
123            DiffEntry::Modified { path, .. } => path,
124        }
125    }
126}
127
128/// 比较两棵 TreeObject，返回所有叶节点（blob/symlink/gitlink）的结构差异列表。
129/// **只比较 OID 和 FileMode，不读取 blob 内容。**
130/// 若需要内容级 diff，请用 `diff_dirs` / `diff_commits`。
131pub fn diff_trees(
132    git_abs: &Path,
133    old: &TreeObject,
134    new: &TreeObject,
135) -> Result<Vec<DiffEntry>> {
136    let mut out = Vec::new();
137    diff_recursive(git_abs, old, new, &PathBuf::new(), &mut out)?;
138    Ok(out)
139}
140
141fn diff_recursive(
142    git_abs: &Path,
143    old: &TreeObject,
144    new: &TreeObject,
145    prefix: &Path,
146    out: &mut Vec<DiffEntry>,
147) -> Result<()> {
148    let mut old_iter = old.entries().iter().peekable();
149    let mut new_iter = new.entries().iter().peekable();
150
151    loop {
152        match (old_iter.peek(), new_iter.peek()) {
153            (None, None) => break,
154            (Some(_), None) => {
155                let (name, entry) = old_iter.next().unwrap();
156                collect_deleted(git_abs, prefix, name, entry, out)?;
157            }
158            (None, Some(_)) => {
159                let (name, entry) = new_iter.next().unwrap();
160                collect_added(git_abs, prefix, name, entry, out)?;
161            }
162            (Some((old_name, _)), Some((new_name, _))) => {
163                match old_name.cmp(new_name) {
164                    Ordering::Less => {
165                        let (name, entry) = old_iter.next().unwrap();
166                        collect_deleted(git_abs, prefix, name, entry, out)?;
167                    }
168                    Ordering::Greater => {
169                        let (name, entry) = new_iter.next().unwrap();
170                        collect_added(git_abs, prefix, name, entry, out)?;
171                    }
172                    Ordering::Equal => {
173                        let (name, old_entry) = old_iter.next().unwrap();
174                        let (_, new_entry) = new_iter.next().unwrap();
175
176                        if old_entry.object_name == new_entry.object_name
177                            && old_entry.file_mode == new_entry.file_mode
178                        {
179                            continue;
180                        }
181
182                        let path = prefix.join(name);
183
184                        if old_entry.file_mode == FileMode::Directory
185                            && new_entry.file_mode == FileMode::Directory
186                        {
187                            let old_sub = TreeObject::read_loose_tree(
188                                git_abs, &old_entry.object_name.to_string()
189                            )?;
190                            let new_sub = TreeObject::read_loose_tree(
191                                git_abs, &new_entry.object_name.to_string()
192                            )?;
193                            diff_recursive(git_abs, &old_sub, &new_sub, &path, out)?;
194                        } else {
195                            // blob↔blob、blob↔dir、dir↔blob
196                            out.push(DiffEntry::Modified {
197                                path,
198                                old_mode: old_entry.file_mode,
199                                new_mode: new_entry.file_mode,
200                                old_oid:  old_entry.object_name.clone(),
201                                new_oid:  new_entry.object_name.clone(),
202                            });
203                        }
204                    }
205                }
206            }
207        }
208    }
209    Ok(())
210}

212fn collect_deleted(
213    git_abs: &Path,
214    prefix: &Path,
215    name: &std::ffi::OsString,
216    entry: &crate::object::TreeEntry,
217    out: &mut Vec<DiffEntry>,
218) -> Result<()> {
219    let path = prefix.join(name);
220    if entry.file_mode == FileMode::Directory {
221        let sub = TreeObject::read_loose_tree(git_abs, &entry.object_name.to_string())?;
222        for (child_name, child_entry) in sub.entries() {
223            collect_deleted(git_abs, &path, child_name, child_entry, out)?;
224        }
225    } else {
226        out.push(DiffEntry::Deleted { path, mode: entry.file_mode, oid: entry.object_name.clone() });
227    }
228    Ok(())
229}
230
231fn collect_added(
232    git_abs: &Path,
233    prefix: &Path,
234    name: &std::ffi::OsString,
235    entry: &crate::object::TreeEntry,
236    out: &mut Vec<DiffEntry>,
237) -> Result<()> {
238    let path = prefix.join(name);
239    if entry.file_mode == FileMode::Directory {
240        let sub = TreeObject::read_loose_tree(git_abs, &entry.object_name.to_string())?;
241        for (child_name, child_entry) in sub.entries() {
242            collect_added(git_abs, &path, child_name, child_entry, out)?;
243        }
244    } else {
245        out.push(DiffEntry::Added { path, mode: entry.file_mode, oid: entry.object_name.clone() });
246    }
247    Ok(())
248}

250// ── subtree_at ────────────────────────────────────────────────────────────────
251
252/// 沿 `path` 的各个分量在 `tree` 中向下导航，返回对应的子 TreeObject。
253/// 路径为空时返回 tree 本身（通过 OID 重读，因为 TreeObject 没有 Clone）。
254/// 中途某个分量不存在、或指向的不是 Directory，返回 `None`。
255pub fn subtree_at(git_abs: &Path, tree: &TreeObject, path: &Path) -> Result<Option<TreeObject>> {
256    let mut current = TreeObject::read_loose_tree(git_abs, &tree.object_name().to_string())?;
257    for component in path.components() {
258        let name = component.as_os_str();
259        match current.entries().get(name) {
260            None => return Ok(None),
261            Some(entry) if entry.file_mode != FileMode::Directory => return Ok(None),
262            Some(entry) => {
263                current = TreeObject::read_loose_tree(git_abs, &entry.object_name.to_string())?;
264            }
265        }
266    }
267    Ok(Some(current))
268}

270// ── 内容级 diff ───────────────────────────────────────────────────────────────

272/// `diff_trees` 的结果加上每个文件的内容 diff
273pub struct FileContentDiff {
274    pub entry: DiffEntry,
275    pub diff:  BlobDiff,
276}
277
278/// 比较两个目录（TreeObject）：先做结构 diff，再对每个变动文件做内容 diff。
279pub fn diff_dirs(
280    git_abs: &Path,
281    old_tree: &TreeObject,
282    new_tree: &TreeObject,
283) -> Result<Vec<FileContentDiff>> {
284    let entries = diff_trees(git_abs, old_tree, new_tree)?;
285    let mut out = Vec::with_capacity(entries.len());
286    for entry in entries {
287        let diff = match &entry {
288            DiffEntry::Added   { oid, .. } => diff_blobs(git_abs, BlobSide::Empty, BlobSide::Stored(oid))?,
289            DiffEntry::Deleted { oid, .. } => diff_blobs(git_abs, BlobSide::Stored(oid), BlobSide::Empty)?,
290            DiffEntry::Modified { old_oid, new_oid, .. } => {
291                diff_blobs(git_abs, BlobSide::Stored(old_oid), BlobSide::Stored(new_oid))?
292            }
293        };
294        out.push(FileContentDiff { entry, diff });
295    }
296    Ok(out)
297}

299/// 比较两个 commit 之间的文件差异：读取各自的 tree 后委托 `diff_dirs`。
300pub fn diff_commits(
301    git_abs: &Path,
302    old_commit_hex: &str,
303    new_commit_hex: &str,
304) -> Result<Vec<FileContentDiff>> {
305    let old_commit = CommitObject::read_loose_commit(git_abs, old_commit_hex)?;
306    let new_commit = CommitObject::read_loose_commit(git_abs, new_commit_hex)?;
307    let old_tree = TreeObject::read_loose_tree(git_abs, &old_commit.tree.to_string())?;
308    let new_tree = TreeObject::read_loose_tree(git_abs, &new_commit.tree.to_string())?;
309    diff_dirs(git_abs, &old_tree, &new_tree)

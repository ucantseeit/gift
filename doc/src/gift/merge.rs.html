29/// 三方归并对单个路径条目的决策结果。
30#[derive(Debug)]
31pub enum MergeEntry {
32    /// 直接采用（不变 / 一侧干净获胜 / 双方改成相同内容）
33    Take {
34        name: OsString,
35        mode: FileMode,
36        oid: ObjectSha,
37    },
38    /// 该条目在合并结果中不存在（一方删除另一方未动，或双方都删除）
39    Delete { name: OsString },
40    /// 两侧做了不兼容改动，需内容级处理
41    Conflict {
42        name: OsString,
43        base: Option<(FileMode, ObjectSha)>,
44        ours: Option<(FileMode, ObjectSha)>,
45        theirs: Option<(FileMode, ObjectSha)>,
46    },
47    /// 三侧都是目录，递归决策结果
48    Subtree {
49        name: OsString,
50        entries: Vec<MergeEntry>,
51    },
52}
53
54type Side = Option<(FileMode, ObjectSha)>;

56/// 三方树归并入口。
57///
58/// 输入三棵 TreeObject（LCA / HEAD / 要合并的分支），返回逐条目的决策列表。
59pub fn merge_trees(
60    git_abs: &Path,
61    base: &TreeObject,
62    ours: &TreeObject,
63    theirs: &TreeObject,
64) -> Result<Vec<MergeEntry>> {
65    let mut out = Vec::new();
66    merge_recursive(git_abs, Some(base), ours, theirs, &mut out)?;
67    Ok(out)
68}

70fn merge_recursive(
71    git_abs: &Path,
72    base: Option<&TreeObject>,
73    ours: &TreeObject,
74    theirs: &TreeObject,
75    out: &mut Vec<MergeEntry>,
76) -> Result<()> {
77    let empty: BTreeMap<OsString, TreeEntry> = BTreeMap::new();
78    let base_entries = base.map(|t| t.entries()).unwrap_or(&empty);
79
80    let mut base_iter = base_entries.iter().peekable();
81    let mut our_iter = ours.entries().iter().peekable();
82    let mut their_iter = theirs.entries().iter().peekable();
83
84    loop {
85        // Clone names up front to avoid holding iterator borrows across the advance calls.
86        let b_name = base_iter.peek().map(|(n, _)| (*n).clone());
87        let o_name = our_iter.peek().map(|(n, _)| (*n).clone());
88        let t_name = their_iter.peek().map(|(n, _)| (*n).clone());
89
90        let name = match [b_name, o_name, t_name].into_iter().flatten().min() {
91            None => break,
92            Some(n) => n,
93        };
94
95        let base_e = advance_if_eq(&mut base_iter, &name);
96        let our_e = advance_if_eq(&mut our_iter, &name);
97        let their_e = advance_if_eq(&mut their_iter, &name);
98
99        out.push(decide(git_abs, name, base_e, our_e, their_e)?);
100    }
101
102    Ok(())
103}

105fn advance_if_eq<'a, I>(iter: &mut std::iter::Peekable<I>, name: &OsString) -> Side
106where
107    I: Iterator<Item = (&'a OsString, &'a TreeEntry)>,
108{
109    if iter.peek().map(|(n, _)| *n) == Some(name) {
110        let (_, entry) = iter.next().unwrap();
111        Some((entry.file_mode, entry.object_name.clone()))
112    } else {
113        None
114    }
115}

117fn sides_equal(a: &Side, b: &Side) -> bool {
118    match (a, b) {
119        (Some((am, ao)), Some((bm, bo))) => am == bm && ao == bo,
120        (None, None) => true,
121        _ => false,
122    }
123}

125fn decide(git_abs: &Path, name: OsString, base: Side, ours: Side, theirs: Side) -> Result<MergeEntry> {
126    // 先借用做比较，再 move 进返回值。
127    let ours_eq_base = sides_equal(&ours, &base);
128    let theirs_eq_base = sides_equal(&theirs, &base);
129    let ours_eq_theirs = sides_equal(&ours, &theirs);
130    let our_is_dir = ours.as_ref().map(|(m, _)| *m) == Some(FileMode::Directory);
131    let their_is_dir = theirs.as_ref().map(|(m, _)| *m) == Some(FileMode::Directory);
132    let base_oid_str = base.as_ref().map(|(_, o)| o.to_string());
133    let our_oid_str = ours.as_ref().map(|(_, o)| o.to_string());
134    let their_oid_str = theirs.as_ref().map(|(_, o)| o.to_string());
135
136    // 双方都删了（或均不存在）
137    if ours.is_none() && theirs.is_none() {
138        return Ok(MergeEntry::Delete { name });
139    }
140
141    // 双方结果一致（含双方都是新增同内容）
142    if ours_eq_theirs {
143        return Ok(match ours {
144            Some((m, o)) => MergeEntry::Take { name, mode: m, oid: o },
145            None => MergeEntry::Delete { name },
146        });
147    }
148
149    // 只有 ours 改动了（theirs 与 base 相同）
150    if theirs_eq_base {
151        return Ok(match ours {
152            Some((m, o)) => MergeEntry::Take { name, mode: m, oid: o },
153            None => MergeEntry::Delete { name },
154        });
155    }
156
157    // 只有 theirs 改动了（ours 与 base 相同）
158    if ours_eq_base {
159        return Ok(match theirs {
160            Some((m, o)) => MergeEntry::Take { name, mode: m, oid: o },
161            None => MergeEntry::Delete { name },
162        });
163    }
164
165    // 双方都改了，且结果不同。
166    if our_is_dir && their_is_dir {
167        // 两侧均为目录 → 递归归并子树。
168        let base_sub = match base_oid_str {
169            Some(ref oid) => Some(TreeObject::read_loose_tree(git_abs, oid)?),
170            None => None,
171        };
172        let our_sub = TreeObject::read_loose_tree(git_abs, &our_oid_str.unwrap())?;
173        let their_sub = TreeObject::read_loose_tree(git_abs, &their_oid_str.unwrap())?;
174        let mut sub_entries = Vec::new();
175        merge_recursive(git_abs, base_sub.as_ref(), &our_sub, &their_sub, &mut sub_entries)?;
176        Ok(MergeEntry::Subtree { name, entries: sub_entries })
177    } else {
178        // 其他情形（blob vs blob / blob vs dir）→ 冲突，交给上层处理。
179        Ok(MergeEntry::Conflict { name, base, ours, theirs })
180    }
181}

185/// merge 结果的顶层类型。
186pub enum MergeOutcome {
187    AlreadyUpToDate,
188    /// 只移动了 HEAD，未创建新 commit
189    FastForward(ObjectSha),
190    /// 成功合并，返回新 merge commit 的 OID
191    Clean(ObjectSha),
192    /// 有冲突，已写入 worktree（含标记）和 index（含 stage 1/2/3），未创建 commit
193    Conflict(Vec<PathBuf>),
194}

196/// 把 `merge_trees` 的决策树落地到 worktree 和 index。
197///
198/// - `Take`：写文件 + add_index stage 0
199/// - `Delete`：删文件 + 从 index 移除
200/// - `Subtree`：递归
201/// - `Conflict`：尝试 diffy 文本合并；成功则写 stage 0，失败则写冲突标记 + stage 1/2/3
202///
203/// 返回有冲突的相对路径列表（为空则合并干净）。
204pub fn merge_apply(
205    git_abs: &Path,
206    worktree: &Path,
207    entries: &[MergeEntry],
208    prefix: &Path,
209    index: &mut IndexFile,
210) -> Result<Vec<PathBuf>> {
211    let mut conflicts = Vec::new();
212
213    for entry in entries {
214        match entry {
215            MergeEntry::Take { name, mode, oid } => {
216                let rel = prefix.join(name);
217                if *mode == FileMode::Directory {
218                    // 整棵子树只在一侧存在，直接 checkout 出来
219                    let abs_dir = worktree.join(&rel);
220                    fs::create_dir_all(&abs_dir)
221                        .with_context(|| format!("mkdir {}", abs_dir.display()))?;
222                    let sub_tree = TreeObject::read_loose_tree(git_abs, &oid.to_string())?;
223                    apply_tree_take(git_abs, worktree, &rel, &sub_tree, index)?;
224                } else {
225                    write_blob_at(git_abs, worktree, &rel, *mode, oid, index)?;
226                }
227            }
228            MergeEntry::Delete { name } => {
229                let rel = prefix.join(name);
230                let abs = worktree.join(&rel);
231                if abs.exists() || fs::symlink_metadata(&abs).is_ok() {
232                    fs::remove_file(&abs)
233                        .with_context(|| format!("remove {}", abs.display()))?;
234                }
235                let path_bytes = index_path_bytes(worktree, &abs)?;
236                index.remove_entry(&path_bytes);
237            }
238            MergeEntry::Subtree { name, entries } => {
239                let sub_prefix = prefix.join(name);
240                let abs_dir = worktree.join(&sub_prefix);
241                fs::create_dir_all(&abs_dir)
242                    .with_context(|| format!("mkdir {}", abs_dir.display()))?;
243                let sub = merge_apply(git_abs, worktree, entries, &sub_prefix, index)?;
244                conflicts.extend(sub);
245            }
246            MergeEntry::Conflict { name, base, ours, theirs } => {
247                let rel = prefix.join(name);
248                apply_conflict(git_abs, worktree, &rel, base, ours, theirs, index, &mut conflicts)?;
249            }
250        }
251    }
252
253    Ok(conflicts)
254}

256/// 把一棵 TreeObject 的全部内容递归写入 worktree + index（用于 Take Directory 分支）。
257fn apply_tree_take(
258    git_abs: &Path,
259    worktree: &Path,
260    prefix: &Path,
261    tree: &TreeObject,
262    index: &mut IndexFile,
263) -> Result<()> {
264    for (name, entry) in tree.entries() {
265        let rel = prefix.join(name);
266        match entry.file_mode {
267            FileMode::Directory => {
268                let abs_dir = worktree.join(&rel);
269                fs::create_dir_all(&abs_dir)?;
270                let sub = TreeObject::read_loose_tree(git_abs, &entry.object_name.to_string())?;
271                apply_tree_take(git_abs, worktree, &rel, &sub, index)?;
272            }
273            mode => {
274                write_blob_at(git_abs, worktree, &rel, mode, &entry.object_name, index)?;
275            }
276        }
277    }
278    Ok(())
279}

281/// 读出 object db 中一个 blob 的原始字节。
282fn read_stored_blob(git_abs: &Path, oid: &ObjectSha) -> Result<Vec<u8>> {
283    let hex = oid.to_string();
284    let mut br = Object::open_loose_object_bufreader(git_abs, &hex)?;
285    BlobObject::read_blob_payload(&mut br, &hex)
286}

288/// 把 `(mode, oid)` 的 blob 写到 `worktree/rel`，并 add_index stage 0。
289fn write_blob_at(
290    git_abs: &Path,
291    worktree: &Path,
292    rel: &Path,
293    mode: FileMode,
294    oid: &ObjectSha,
295    index: &mut IndexFile,
296) -> Result<()> {
297    let abs = worktree.join(rel);
298    if let Some(p) = abs.parent() {
299        fs::create_dir_all(p)?;
300    }
301    let payload = read_stored_blob(git_abs, oid)?;
302
303    match mode {
304        FileMode::SymbolicLink => {
305            if abs.exists() || fs::symlink_metadata(&abs).is_ok() {
306                fs::remove_file(&abs)?;
307            }
308            let target = OsString::from_vec(payload);
309            symlink(Path::new(&target), &abs)
310                .with_context(|| format!("symlink {}", abs.display()))?;
311        }
312        _ => {
313            fs::write(&abs, &payload)
314                .with_context(|| format!("write {}", abs.display()))?;
315            let exec = mode == FileMode::ExecRegularFile;
316            let mut perms = fs::metadata(&abs)?.permissions();
317            perms.set_mode(if exec { 0o755 } else { 0o644 });
318            fs::set_permissions(&abs, perms)?;
319        }
320    }
321
322    let md = fs::symlink_metadata(&abs)?;
323    let path_bytes = index_path_bytes(worktree, &abs)?;
324    add_index(index, &md, path_bytes, oid.clone())?;
325    Ok(())
326}

328/// 对冲突条目做内容合并：文本用 diffy，二进制直接写 ours。
329fn apply_conflict(
330    git_abs: &Path,
331    worktree: &Path,
332    rel: &Path,
333    base: &Option<(FileMode, ObjectSha)>,
334    ours: &Option<(FileMode, ObjectSha)>,
335    theirs: &Option<(FileMode, ObjectSha)>,
336    index: &mut IndexFile,
337    conflicts: &mut Vec<PathBuf>,
338) -> Result<()> {
339    let abs = worktree.join(rel);
340    if let Some(p) = abs.parent() {
341        fs::create_dir_all(p)?;
342    }
343
344    let base_bytes  = match base   { Some((_, o)) => read_stored_blob(git_abs, o)?, None => Vec::new() };
345    let our_bytes   = match ours   { Some((_, o)) => read_stored_blob(git_abs, o)?, None => Vec::new() };
346    let their_bytes = match theirs { Some((_, o)) => read_stored_blob(git_abs, o)?, None => Vec::new() };
347
348    let is_binary = [&base_bytes, &our_bytes, &their_bytes]
349        .iter()
350        .any(|b| b.contains(&0u8));
351
352    let path_bytes = index_path_bytes(worktree, &abs)?;
353
354    if !is_binary {
355        if let (Ok(b), Ok(o), Ok(t)) = (
356            std::str::from_utf8(&base_bytes),
357            std::str::from_utf8(&our_bytes),
358            std::str::from_utf8(&their_bytes),
359        ) {
360            if let Ok(merged) = diffy::merge(b, o, t) {
361                // 干净合并：写文件 + 写 blob + add_index stage 0
362                fs::write(&abs, merged.as_bytes())?;
363                let (merged_oid, obj_content) = hash_object(&abs)?;
364                write_hash_object(git_abs, &merged_oid, &obj_content)?;
365                let md = fs::symlink_metadata(&abs)?;
366                add_index(index, &md, path_bytes, merged_oid)?;
367                return Ok(());
368            } else {
369                // 有冲突标记：diffy 已把标记写进返回值，写入文件（Err 的内容就是带标记的字符串）
370                // 重新调一次拿到带标记的结果
371                let with_markers = diffy::merge(b, o, t).unwrap_err();
372                fs::write(&abs, with_markers.as_bytes())?;
373            }
374        }
375        // UTF-8 解码失败时也走到这里（作为二进制冲突处理）
376    }
377
378    // 二进制或文本冲突：写 ours 内容到 worktree，index 写 stage 1/2/3
379    if is_binary || our_bytes.is_empty() {
380        fs::write(&abs, &our_bytes)?;
381    }
382    // （文本冲突时文件已在上方写入带标记内容）
383
384    insert_conflict_entries(
385        index,
386        path_bytes,
387        base.clone(),
388        ours.clone(),
389        theirs.clone(),
390    );
391    conflicts.push(rel.to_path_buf());
392    Ok(())
393}

397/// 将分支名或 40 位 hex OID 解析成 `ObjectSha`。
398///
399/// - 40 位 hex → 直接解码为 SHA1 OID
400/// - 其他字符串 → 视为本地分支名，读取 `refs/heads/<name>`
401fn resolve_to_commit_oid(git_abs: &Path, target: &str) -> Result<ObjectSha> {
402    let t = target.trim();
403    if t.len() == 40 && t.chars().all(|c| c.is_ascii_hexdigit()) {
404        // 直接当作 OID
405        let bytes: [u8; 20] = hex::decode(t)
406            .context("OID hex decode")?
407            .try_into()
408            .map_err(|_| anyhow::anyhow!("OID must be 20 bytes"))?;
409        return Ok(ObjectSha::SHA1(bytes));
410    }
411    // 否则当作分支名，从 refs/heads/ 读 OID
412    let ref_path = git_abs.join("refs").join("heads").join(t);
413    ensure!(
414        ref_path.exists(),
415        "branch '{}' not found (tried {})",
416        t,
417        ref_path.display()
418    );
419    let hex = fs::read_to_string(&ref_path)
420        .with_context(|| format!("read ref {}", ref_path.display()))?;
421    let hex = hex.trim();
422    let bytes: [u8; 20] = hex::decode(hex)
423        .context("branch ref hex decode")?
424        .try_into()
425        .map_err(|_| anyhow::anyhow!("branch ref OID must be 20 bytes"))?;
426    Ok(ObjectSha::SHA1(bytes))
427}

429/// 三方 merge 入口（resolve 策略，不做 rename 检测）。
430///
431/// `target` 可以是 40 位 hex OID 或本地分支名（`resolve_to_commit_oid` 处理解析）。
432///
433/// 返回值含义：
434/// - `AlreadyUpToDate`：their 是我方祖先，无需操作
435/// - `FastForward`：我方是 their 祖先，直接移动 HEAD
436/// - `Clean(oid)`：成功创建 merge commit，返回新 commit OID
437/// - `Conflict(paths)`：有冲突，冲突文件写入 worktree（含 `<<<<<<<` 标记），
438///    index 含 stage 1/2/3，`MERGE_HEAD` / `MERGE_MSG` 已写入，
439///    用户解决后可 `gift add` + `gift commit` 完成合并
440pub fn merge(
441    worktree: &Path,
442    git_abs: &Path,
443    target: &str,
444    author: CommitIdentity,
445    committer: CommitIdentity,
446    message: &str,
447) -> Result<MergeOutcome> {
448    // 1. 解析 their OID（支持分支名或 hex OID）
449    let their_oid = resolve_to_commit_oid(git_abs, target)?;
450
451    // 2. 读 HEAD → our OID
452    let head = Head::read(git_abs)?;
453    let our_oid = head.current_commit(git_abs)?;
454
455    if our_oid == their_oid {
456        return Ok(MergeOutcome::AlreadyUpToDate);
457    }
458
459    // 记录合并前的 HEAD，供用户 `reset --hard ORIG_HEAD` 撤销
460    fs::write(git_abs.join("ORIG_HEAD"), format!("{}\n", our_oid.to_string()))?;
461
462    // 3. 读两侧 commit / tree
463    let our_commit   = CommitObject::read_loose_commit(git_abs, &our_oid.to_string())?;
464    let their_commit = CommitObject::read_loose_commit(git_abs, &their_oid.to_string())?;
465    let our_tree   = TreeObject::read_loose_tree(git_abs, &our_commit.tree.to_string())?;
466    let their_tree = TreeObject::read_loose_tree(git_abs, &their_commit.tree.to_string())?;
467
468    // 4. 找 merge base
469    let base_oid_opt = find_merge_base(git_abs, &our_oid, &their_oid)?;
470
471    match &base_oid_opt {
472        Some(base_oid) if base_oid == &their_oid => {
473            return Ok(MergeOutcome::AlreadyUpToDate);
474        }
475        Some(base_oid) if base_oid == &our_oid => {
476            // Fast-forward：ours 与 base 相同，theirs 的所有改动直接 Take
477            let entries = merge_trees(git_abs, &our_tree, &our_tree, &their_tree)?;
478            let mut index = IndexFile::empty(2);
479            merge_apply(git_abs, worktree, &entries, Path::new(""), &mut index)?;
480            write_index_file(&git_abs.join("index"), &index)?;
481            head.record_new_commit(worktree, git_abs, &their_oid)?;
482            return Ok(MergeOutcome::FastForward(their_oid));
483        }
484        None => bail!("refusing to merge unrelated histories (no common ancestor)"),
485        _ => {} // 正常三方合并
486    }
487
488    // 5. 三方合并
489    let base_oid   = base_oid_opt.unwrap();
490    let base_commit = CommitObject::read_loose_commit(git_abs, &base_oid.to_string())?;
491    let base_tree  = TreeObject::read_loose_tree(git_abs, &base_commit.tree.to_string())?;
492
493    let entries = merge_trees(git_abs, &base_tree, &our_tree, &their_tree)?;
494
495    let mut index = IndexFile::empty(2);
496    let conflict_paths = merge_apply(git_abs, worktree, &entries, Path::new(""), &mut index)?;
497    write_index_file(&git_abs.join("index"), &index)?;
498
499    if conflict_paths.is_empty() {
500        // 写 tree、创建 merge commit（两个 parent）
501        let tree_root  = TreeNode::from_index_file(&index)?;
502        let tree_entry = tree_root.write_tree_return_entry(git_abs, true);
503        let commit = CommitObject::new(
504            tree_entry.object_name,
505            vec![our_oid, their_oid],
506            author,
507            committer,
508            Vec::new(),
509            message.as_bytes().to_vec(),
510        );
511        let commit_oid = commit_tree(git_abs, &commit)?;
512        head.record_new_commit(worktree, git_abs, &commit_oid)?;
513        Ok(MergeOutcome::Clean(commit_oid))
514    } else {
515        // 写 merge 状态文件：供后续 `gift commit` 使用
516        fs::write(git_abs.join("MERGE_HEAD"), format!("{}\n", their_oid.to_string()))?;
517        fs::write(
518            git_abs.join("MERGE_MSG"),
519            message.to_string(),
520        )?;
521        Ok(MergeOutcome::Conflict(conflict_paths))
522    }
523}

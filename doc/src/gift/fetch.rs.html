9/// 一条成功更新的引用，便于打印 fetch 摘要。
10#[derive(Debug)]
11pub struct FetchedRef {
12    pub remote_name: String, // 远端原始 refname，如 refs/heads/main
13    pub local_name: String,  // 本地落点，如 refs/remotes/origin/main
14    pub sha: String,
15}
16
17/// 把广告里的引用映射成本地引用并写盘：
18///   - `refs/heads/<x>` → `refs/remotes/<remote>/<x>`（远程跟踪分支）
19///   - `refs/tags/<x>`  → `refs/tags/<x>`（标签是全局的，直接照搬）
20///   - 其它命名空间（`refs/pull/...` 等）跳过
21///
22/// 同时（重）写 `FETCH_HEAD`，每行 `"<sha>\t\t<描述> of <url>"`。
23/// `FETCH_HEAD` 不是逐字节兼容 git 的格式（我们没实现 merge 标记），
24/// 仅作为“上次 fetch 了什么”的可读记录。
25fn write_fetched_refs(
26    git_abs: &Path,
27    remote: &str,
28    url: &str,
29    adv: &RefAdvertisement,
30) -> std::io::Result<Vec<FetchedRef>> {
31    let mut fetched = Vec::new();
32    let mut fetch_head = String::new();
33
34    for r in &adv.refs {
35        if r.name == "HEAD" || r.name.ends_with("^{}") {
36            continue; // HEAD 不建跟踪引用；peeled(^{}) 行不写成 ref
37        }
38
39        // 决定本地落点（相对 git_abs 的路径）
40        let (local_rel, desc) = if let Some(branch) = r.name.strip_prefix("refs/heads/") {
41            (format!("refs/remotes/{remote}/{branch}"), format!("branch '{branch}'"))
42        } else if let Some(tag) = r.name.strip_prefix("refs/tags/") {
43            (r.name.clone(), format!("tag '{tag}'"))
44        } else {
45            continue; // 其它命名空间不跟踪
46        };
47
48        let path = git_abs.join(&local_rel);
49        if let Some(parent) = path.parent() {
50            fs::create_dir_all(parent)?;
51        }
52        fs::write(&path, format!("{}\n", r.sha))?; // ref 文件内容就是 40 位 sha
53
54        fetch_head.push_str(&format!("{}\t\t{} of {}\n", r.sha, desc, url));
55        fetched.push(FetchedRef {
56            remote_name: r.name.clone(),
57            local_name: local_rel,
58            sha: r.sha.clone(),
59        });
60    }
61
62    fs::write(git_abs.join("FETCH_HEAD"), fetch_head)?;
63    Ok(fetched)
64}

66/// 落盘半部（与 `clone_to_disk` 对位）：把内存里的对象写进**已存在**仓库的
67/// 对象库，并更新远程跟踪引用 + FETCH_HEAD。不碰 HEAD / 本地分支 / 工作区。
68///
69/// `git_abs` 是已存在仓库的 git 目录绝对路径（如 `.../.gift`；测试里传 `.git`）。
70pub fn fetch_to_disk(
71    git_abs: &Path,
72    remote: &str,
73    url: &str,
74    objects: &[PackedObject],
75    adv: &RefAdvertisement,
76) -> std::io::Result<Vec<FetchedRef>> {
77    // 1) 对象落盘到现有对象库（已存在的会被 write_loose_object 跳过）
78    for o in objects {
79        write_loose_object(git_abs, o)?;
80    }
81    // 2) 远程跟踪引用 + FETCH_HEAD
82    write_fetched_refs(git_abs, remote, url, adv)
83}

85/// 判断字符串是否是 40 位十六进制（一个 SHA1 oid）。
86fn is_hex40(s: &str) -> bool {
87    s.len() == 40 && s.bytes().all(|b| b.is_ascii_hexdigit())
88}
89
90/// 递归收集 `dir`（通常是 `refs/`）下所有直接 ref 文件里的 40 位 sha。
91/// symref 文件（内容 `ref: ...`）会被 `is_hex40` 自然过滤掉。
92fn collect_refs_recursive(dir: &Path, set: &mut HashSet<String>) -> io::Result<()> {
93    let rd = match fs::read_dir(dir) {
94        Ok(rd) => rd,
95        Err(e) if e.kind() == io::ErrorKind::NotFound => return Ok(()),
96        Err(e) => return Err(e),
97    };
98    for ent in rd {
99        let path = ent?.path();
100        if path.is_dir() {
101            collect_refs_recursive(&path, set)?;
102        } else if let Ok(s) = fs::read_to_string(&path) {
103            let s = s.trim();
104            if is_hex40(s) {
105                set.insert(s.to_string());
106            }
107        }
108    }
109    Ok(())
110}
111
112/// 收集本地“已有对象”的代表集合，作为协商用的 have 列表：
113/// 所有本地 ref（refs/heads、refs/remotes、refs/tags …）的 tip，外加 detached HEAD。
114/// 服务器知道某个 have 后，即认为我们已拥有它及其可达的一切，从而只发新对象。
115fn collect_local_haves(git_abs: &Path) -> io::Result<Vec<String>> {
116    let mut set = HashSet::new();
117    collect_refs_recursive(&git_abs.join("refs"), &mut set)?;
118    if let Ok(head) = fs::read_to_string(git_abs.join("HEAD")) {
119        let h = head.trim();
120        if is_hex40(h) {
121            set.insert(h.to_string()); // detached HEAD 直接是 sha
122        }
123    }
124    Ok(set.into_iter().collect())
125}

127/// 顶层 fetch：复用 clone 的网络件套做增量协商，再走 fetch 专属的落盘。
128///
129/// 本项目没有 remote 配置系统，所以远端用 `url` 直接给出，`remote` 是要写到
130/// `refs/remotes/<remote>/*` 的名字（CLI 默认 "origin"）。
131pub fn fetch(git_abs: &Path, url: &str, remote: &str) -> Result<(), anyhow::Error> {
132    let haves = collect_local_haves(git_abs)?; // 本地已有 → 协商，避免重复下载
133    let adv = discover_refs(url)?; // 复用：远端 ref 广告（refs + caps + HEAD）
134    let pack = fetch_pack(url, &adv, &haves)?; // 复用：发 want+have、side-band 解复用 → pack
135    let objects = parse_pack_thin(&pack, git_abs)?; // thin-pack：外部基从本地对象库增厚
136
137    let fetched = fetch_to_disk(git_abs, remote, url, &objects, &adv)?;
138
139    // 打印一个朴素的 fetch 摘要
140    println!("From {url}");
141    for f in &fetched {
142        let short = &f.sha[..f.sha.len().min(7)];
143        println!("   {short}  {} -> {}", f.remote_name, f.local_name);
144    }
145    Ok(())
146}

151    use std::process::Command;

153    fn run(c: &mut Command) {
154        assert!(c.status().unwrap().success(), "命令失败: {c:?}");

157    /// 离线验证落盘半部：源仓库当“服务器”，把对象 + 广告 fetch 进一个**已存在**
158    /// 的目标仓库，断言：① 对象进了对象库；② refs/remotes/origin/* 正确建立；
159    /// ③ HEAD 与本地分支**没有**被动过。
160    #[test]
161    fn fetch_to_disk_updates_tracking_refs_only() {
162        let env = [
163            ("GIT_AUTHOR_NAME", "t"), ("GIT_AUTHOR_EMAIL", "t@t"),
164            ("GIT_COMMITTER_NAME", "t"), ("GIT_COMMITTER_EMAIL", "t@t"),
165        ];
166
167        // ── 源仓库（当服务器）：1 提交 + 1 标签，repack 成单 pack ──
168        let src = std::env::temp_dir().join(format!("fetchsrc_{}", std::process::id()));
169        let _ = fs::remove_dir_all(&src);
170        run(Command::new("git").args(["init", "-q"]).arg(&src));
171        fs::write(src.join("README.md"), "hello\n").unwrap();
172        run(Command::new("git").arg("-C").arg(&src).args(["add", "."]));
173        run(Command::new("git").arg("-C").arg(&src).args(["commit", "-q", "-m", "first"]).envs(env));
174        run(Command::new("git").arg("-C").arg(&src).args(["tag", "v1"]).envs(env));
175        run(Command::new("git").arg("-C").arg(&src).args(["repack", "-q", "-a", "-d"]));
176
177        // 解析广告（复用 clone 的 parse_advertisement，经 upload-pack）
178        use crate::get_packfile_by_network::parse_advertisement;
179        let adv_bytes = Command::new("git").args(["upload-pack", "--advertise-refs"]).arg(&src)
180            .output().unwrap().stdout;
181        let mut cur: &[u8] = &adv_bytes;
182        let adv = parse_advertisement(&mut cur).unwrap();
183        // 默认分支名随 git 配置可能是 master/main，从广告推导而非硬编码
184        let head_branch = adv.head_target.clone().expect("应有 symref=HEAD 指向默认分支");
185        let branch_short = head_branch.strip_prefix("refs/heads/").unwrap().to_string();
186
187        // 解析 pack（这里是自洽 pack，用普通 parse_pack 即可）
188        let packdir = src.join(".git/objects/pack");
189        let pack = fs::read_dir(&packdir).unwrap().filter_map(|e| e.ok().map(|e| e.path()))
190            .find(|p| p.extension().map_or(false, |x| x == "pack")).unwrap();
191        let objects = crate::parse_packfile::parse_pack(&fs::read(&pack).unwrap()).unwrap();
192
193        // ── 目标仓库：一个**已存在**的、互不相干的空仓库 ──
194        let dst = std::env::temp_dir().join(format!("fetchdst_{}", std::process::id()));
195        let _ = fs::remove_dir_all(&dst);
196        run(Command::new("git").args(["init", "-q"]).arg(&dst));
197        let dst_git = dst.join(".git");
198        let head_before = fs::read(dst_git.join("HEAD")).unwrap();
199
200        // fetch 落盘（测试里 git 目录用 ".git" 以便直接用 git 验证）
201        let fetched = fetch_to_disk(&dst_git, "origin", "../src", &objects, &adv).unwrap();
202        assert!(!fetched.is_empty(), "应至少 fetch 到一条 ref");
203
204        // ① 对象进了对象库：源的每个 ref tip 在目标里都能 cat-file -e
205        for r in &adv.refs {
206            if r.name.ends_with("^{}") { continue; }
207            let ok = Command::new("git").arg("-C").arg(&dst)
208                .args(["cat-file", "-e", &r.sha]).status().unwrap().success();
209            assert!(ok, "对象 {} 应已写入目标对象库", r.sha);
210        }
211
212        // ② 远程跟踪分支建立且 sha 正确
213        let head_sha = adv.sha_of(&head_branch).expect("源应有默认分支");
214        let track = dst_git.join(format!("refs/remotes/origin/{branch_short}"));
215        assert_eq!(fs::read_to_string(&track).unwrap().trim(), head_sha,
216                   "refs/remotes/origin/{branch_short} 应指向源默认分支");
217        // 标签照搬
218        assert!(dst_git.join("refs/tags/v1").exists(), "标签应照搬到 refs/tags/v1");
219        // FETCH_HEAD 写出来了
220        assert!(dst_git.join("FETCH_HEAD").exists(), "应写出 FETCH_HEAD");
221
222        // ③ HEAD 没被动；本地分支不存在（fetch 不建本地分支）
223        assert_eq!(fs::read(dst_git.join("HEAD")).unwrap(), head_before, "fetch 不应改 HEAD");
224        assert!(!dst_git.join(&head_branch).exists(), "fetch 不应创建本地分支");
225
226        let _ = fs::remove_dir_all(&src);
227        let _ = fs::remove_dir_all(&dst);
228    }

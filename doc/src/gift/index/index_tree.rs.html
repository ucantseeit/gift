34pub enum TreeNode {
35    Blob(BlobLeaf),
36    Tree(BTreeMap<OsString, TreeNode>)
37}
38
39impl TreeNode {
40    /// 把 blob 沿 `parent_dir_iter` 插入：沿途各级 Tree 的 children map 会被就地修改或新建；本节点必须是 `Tree`，否则 bail!
41    pub fn insert_blob(
42        &mut self, 
43        parent_dir_iter: &mut Components<'_>, 
44        blob_file_name: OsString, 
45        blob: BlobLeaf
46    ) -> Result<()> {
47        let TreeNode::Tree(tree) = self else {
48            bail!("Blob无法被合并")
49        };
50        TreeNode::insert_blob_into_children_map(tree, parent_dir_iter, blob_file_name, blob)
51    }
52
53    /// 递归把子树写入 loose object，返回本节点的 TreeEntry（Blob 直接返回，Tree 有写磁盘副作用）
54    pub fn write_tree_return_entry(&self, git_abs: &Path, is_sha1: bool) -> TreeEntry {
55        match self {
56            TreeNode::Blob(b) => {
57                TreeEntry{file_mode: b.file_mode, object_name: b.object_name.clone()}
58            }
59            TreeNode::Tree(children) => {
60                let mut entries: BTreeMap<OsString, TreeEntry> = BTreeMap::new();
61                for (file_name, child) in children {
62                    let entry = child.write_tree_return_entry(git_abs.as_ref(), is_sha1);
63                    entries.insert(file_name.clone(), entry);
64                };
65                let content = TreeObject::entries_to_binary(entries, is_sha1);
66                let hash: [u8; 20] = Sha1::digest(&content).try_into().unwrap();
67                let object_name = ObjectSha::SHA1(hash);
68                write_hash_object(git_abs, &object_name, &content).unwrap();
69                TreeEntry { file_mode: FileMode::Directory, object_name }
70            }
71        }
72    }
73
74    /// 把 index 里记录的扁平文件路径还原成与工作区目录层级一致的 TreeNode 树；
75    /// index 条目已按路径字典序排列（BTreeMap 保证），逐条插入即可得到正确的结构
76    pub fn from_index_file(index_file: &IndexFile) -> Result<TreeNode> {
77        let mut result = TreeNode::Tree( BTreeMap::new() );
78        for entry in index_file.entries().filter(|e| e.merge_stage() == 0) {
79            let path = entry.decode_entry_path();
80            let Some(file_name) = path.file_name() else {
81                bail!("index文件中存在没有file_name的entry");
82            };
83            let blob = BlobLeaf { 
84                // file_name: file_name.to_os_string(),
85                file_mode: entry.file_mode(), 
86                object_name: entry.obj_name().clone() 
87            };
88
89            let parent_path = path.parent().unwrap_or_else(|| Path::new(""));
90            let mut parent_dir_iter = parent_path.components();
91            result.insert_blob(&mut parent_dir_iter, file_name.to_owned(), blob).unwrap();
92        }
93
94        Ok(result)
95    }
96
97    /// 每次递归调用从 `parent_dir_iter` 取一个目录分量，
98    /// 在 `map` 中新建或深入对应子树，
99    /// 分量耗尽时将 blob 作为值写入 map
100    fn insert_blob_into_children_map(
101        map: &mut BTreeMap<OsString, TreeNode>,
102        parent_dir_iter: &mut Components<'_>,
103        blob_file_name: OsString,
104        blob: BlobLeaf,
105    ) -> Result<()> {
106        // 提取路径
107        let Some(child_name) = parent_dir_iter
108            .next()
109            .map(|c| c.as_os_str().to_owned())
110        else {
111            // 如果 parent_dir_iter 已经为空, 直接写入 blob 后return
112            map.insert(blob_file_name, TreeNode::Blob(blob));
113            return Ok(());
114        };
115
116        // 在 map 里新插入子文件夹 (如果这个文件夹还不存在)
117        match map.entry(child_name) {
118            btree_map::Entry::Occupied(mut e) => {
119                e.get_mut().insert_blob(parent_dir_iter, blob_file_name, blob)?;
120            }
121            btree_map::Entry::Vacant(e) => {
122                let mut child_map = BTreeMap::new();
123                Self::insert_blob_into_children_map(&mut child_map, parent_dir_iter, blob_file_name, blob)?;
124                e.insert(TreeNode::Tree(child_map));
125            }
126        }
127        Ok(())
128    }

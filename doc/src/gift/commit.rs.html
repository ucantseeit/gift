1//! 根据暂存区（index）创建一次提交并更新 `HEAD` / 分支 tip。
2//!
3//! 流程（与实现计划一致）：
4//! 1. 解析 `git_abs` 下的 index，用 `IndexRootTree::from_index_file` 构建内存树，再 `write_tree` 写入 loose tree 并得到根 tree 的 OID。
5//! 2. 读取 `HEAD`：区分 symbolic（分支）与 detached（`HEAD` 内直接为 40 位 hex）。
6//! 3. 父 commit：
7//!    - detached（`TargetCommit`）：父为 `HEAD` 中的 OID（须为 `commit` 类型）。
8//!    - 分支（`TargetBranch`）：若 tip ref 文件不存在则无父（初始提交）；若存在则读一行 OID，校验为 SHA1 且对象为 `commit`，则父为该 OID。
9//! 4. 构造 `CommitObject`：author / committer 由调用方传入（可从环境变量、`.git/config` 等在上层解析）；本模块再 `commit_tree` 写入 loose commit。
10//! 5. 更新 `HEAD`：非 detached 只更新分支 tip ref；detached 将 `HEAD` 改为新 OID（`Head::record_new_commit`）。
11
12use std::fs;
13use std::path::Path;
14
15use anyhow::{Context, Result, bail, ensure};
16use crate::head::Head;
17use crate::index::{index_tree::TreeNode, index_file::parse_index_file};
18use crate::object::{commit_tree, Object, CommitIdentity, CommitObject, ObjectSha};

20/// 使用当前 index 创建一次提交：写 tree、写 commit、按 `HEAD` 形态更新引用。
21///
22/// - `worktree`：工作区根路径（与 `Head::read`、`read_ref` 约定一致）。
23/// - `git_abs`：git仓库绝对路径
24/// - `author` / `committer`：由上层提供（例如 [`crate::commit_identity::identities_from_git_env`] 或从 config 读取后再构造 [`CommitIdentity`]）。
25/// - `commit_message`：提交说明（写入 commit 对象 body；若末尾无换行会补一个 `\n`，与常见 Git 行为一致）。
26///
27/// 返回新产生的 commit OID（SHA1）。
28pub fn commit(
29    worktree: &Path,
30    git_abs: &Path,
31    author: CommitIdentity,
32    committer: CommitIdentity,
33    commit_message: Option<String>,
34) -> Result<ObjectSha> {
35    let index_path = git_abs.join("index");
36    let index_file = parse_index_file(&index_path)
37        .with_context(|| format!("parse index {}", index_path.display()))?;
38    let index_tree = TreeNode::from_index_file(&index_file)
39        .context("index -> IndexRootTree")?;
40    let tree_oid = index_tree.write_tree_return_entry(git_abs, true).object_name;
41
42    let head = Head::read(git_abs).context("read HEAD")?;
43    let parents = resolve_parents(&git_abs, &head)?;
44
45    // message 优先取调用方传入；无则回退到 MERGE_MSG（merge 冲突解决后 commit）
46    let resolved_message = match commit_message {
47        Some(m) => m,
48        None => {
49            let p = git_abs.join("MERGE_MSG");
50            ensure!(p.exists(), "no -m given and MERGE_MSG not found");
51            fs::read_to_string(&p)
52                .context("read MERGE_MSG")?
53                .trim_end()
54                .to_string()
55        }
56    };
57
58    let mut message = resolved_message.into_bytes();
59    if !message.ends_with(b"\n") {
60        message.push(b'\n');
61    }
62
63    let commit_obj = CommitObject::new(
64        tree_oid,
65        parents,
66        author,
67        committer,
68        Vec::new(),
69        message,
70    );
71
72    let new_oid = commit_tree(&git_abs, &commit_obj).context("write commit object")?;
73    head
74        .record_new_commit(worktree, git_abs, &new_oid)
75        .context("update HEAD / branch ref")?;
76
77    // merge 完成后清理状态文件（文件不存在时静默忽略）
78    let _ = fs::remove_file(git_abs.join("MERGE_HEAD"));
79    let _ = fs::remove_file(git_abs.join("MERGE_MSG"));
80
81    Ok(new_oid)
82}

84/// 解析当前 HEAD 对应的 parent commit 列表，并附加 MERGE_HEAD（若存在）。
85fn resolve_parents(git_abs: &Path, head: &Head) -> Result<Vec<ObjectSha>> {
86    let mut parents = match head {
87        Head::TargetCommit(oid) => {
88            let mut reader = Object::open_loose_object_bufreader(git_abs, &oid.to_string())?;
89            Object::ensure_loose_object_kind(&mut reader, "commit", "detached HEAD")?;
90            vec![oid.clone()]
91        }
92        Head::TargetBranch(symref) => {
93            let branch_ref_abs = git_abs.join(&symref.ref_path);
94
95            // git init 后 tip ref 尚不存在 → 初始提交，无 parent
96            if !branch_ref_abs.exists() {
97                Vec::new()
98            } else {
99                let content = fs::read_to_string(&branch_ref_abs)
100                    .with_context(|| format!("read branch ref {}", branch_ref_abs.display()))?;
101                let line = content.trim();
102
103                if line.is_empty() {
104                    bail!("branch ref file is empty: {}", branch_ref_abs.display());
105                }
106                if line.lines().nth(1).is_some() {
107                    bail!("branch ref must be a single line: {}", branch_ref_abs.display());
108                }
109                if line.len() != 40 || !line.chars().all(|c| c.is_ascii_hexdigit()) {
110                    bail!("branch ref must be 40 hex chars: {}", branch_ref_abs.display());
111                }
112
113                let bytes: [u8; 20] = hex::decode(line)
114                    .with_context(|| format!("decode ref {}", branch_ref_abs.display()))?
115                    .try_into()
116                    .map_err(|v: Vec<u8>| anyhow::anyhow!("ref oid length {}", v.len()))?;
117                let oid = ObjectSha::SHA1(bytes);
118                let mut reader = Object::open_loose_object_bufreader(git_abs, &oid.to_string())?;
119                Object::ensure_loose_object_kind(&mut reader, "commit", "branch tip")?;
120                vec![oid]
121            }
122        }
123    };
124
125    // merge 进行中时追加 MERGE_HEAD 作为第二 parent
126    let merge_head_path = git_abs.join("MERGE_HEAD");
127    if merge_head_path.exists() {
128        let hex = fs::read_to_string(&merge_head_path)
129            .context("read MERGE_HEAD")?;
130        let hex = hex.trim();
131        let bytes: [u8; 20] = hex::decode(hex)
132            .context("MERGE_HEAD hex decode")?
133            .try_into()
134            .map_err(|_| anyhow::anyhow!("MERGE_HEAD OID is not 20 bytes"))?;
135        let oid = ObjectSha::SHA1(bytes);
136        let mut reader = Object::open_loose_object_bufreader(git_abs, &oid.to_string())?;
137        Object::ensure_loose_object_kind(&mut reader, "commit", "MERGE_HEAD")?;
138        parents.push(oid);
139    }
140
141    Ok(parents)
142}

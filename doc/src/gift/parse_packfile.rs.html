18pub enum Kind { Commit, Tree, Blob, Tag }

20impl Kind {
21    fn from_type(t: u8) -> Option<Kind> {
22        match t { 1 => Some(Kind::Commit), 2 => Some(Kind::Tree), 3 => Some(Kind::Blob), 4 => Some(Kind::Tag), _ => None }
23    }
24    pub fn name(self) -> &'static str {
25        match self { Kind::Commit => "commit", Kind::Tree => "tree", Kind::Blob => "blob", Kind::Tag => "tag" }
26    }

46fn read_u8(data: &[u8], pos: &mut usize) -> io::Result<u8> {
47    let b = *data.get(*pos).ok_or_else(|| eof("pack 提前结束"))?;
48    *pos += 1;
49    Ok(b)
50}

52fn eof(msg: &str) -> io::Error { io::Error::new(io::ErrorKind::UnexpectedEof, msg.to_string()) }
53fn bad(msg: impl Into<String>) -> io::Error { io::Error::new(io::ErrorKind::InvalidData, msg.into()) }

55/// 对象条目的变长头：bit6-4=类型，其余位拼出“解压后大小”。
56fn read_type_and_size(data: &[u8], pos: &mut usize) -> io::Result<(u8, usize)> {
57    let b = read_u8(data, pos)?;
58    let typ = (b >> 4) & 0x07;
59    let mut size = (b & 0x0f) as usize;
60    let mut shift = 4;
61    let mut more = b & 0x80 != 0;
62    while more {
63        let b = read_u8(data, pos)?;
64        size |= ((b & 0x7f) as usize) << shift;
65        shift += 7;
66        more = b & 0x80 != 0;
67    }
68    Ok((typ, size))
69}

71/// ofs-delta 的“向前负偏移”（特殊变长编码）。
72fn read_ofs_offset(data: &[u8], pos: &mut usize) -> io::Result<usize> {
73    let mut b = read_u8(data, pos)?;
74    let mut offset = (b & 0x7f) as usize;
75    while b & 0x80 != 0 {
76        b = read_u8(data, pos)?;
77        offset = ((offset + 1) << 7) | (b & 0x7f) as usize;
78    }
79    Ok(offset)
80}

82/// delta 内部的大小变长整数（小端 base-128）。
83fn read_varint(data: &[u8], pos: &mut usize) -> io::Result<usize> {
84    let mut size = 0usize;
85    let mut shift = 0;
86    loop {
87        let b = read_u8(data, pos)?;
88        size |= ((b & 0x7f) as usize) << shift;
89        shift += 7;
90        if b & 0x80 == 0 { break; }
91    }
92    Ok(size)
93}

95/// 解压一个 zlib 流。边界由解压器自己定，返回 (解压内容, 消耗的压缩字节数)。
96fn inflate(input: &[u8], expected: usize) -> io::Result<(Vec<u8>, usize)> {
97    let mut d = ZlibDecoder::new(input);
98    let mut out = Vec::with_capacity(expected);
99    d.read_to_end(&mut out)?;
100    Ok((out, d.total_in() as usize))
101}

103/// 套用 delta 指令，从 base 重建出目标内容。
104fn apply_delta(base: &[u8], delta: &[u8]) -> io::Result<Vec<u8>> {
105    let mut pos = 0;
106    let base_size = read_varint(delta, &mut pos)?;
107    if base_size != base.len() {
108        return Err(bad("delta 声明的基大小与基对象不符"));
109    }
110    let target_size = read_varint(delta, &mut pos)?;
111    let mut out = Vec::with_capacity(target_size);
112
113    while pos < delta.len() {
114        let cmd = delta[pos];
115        pos += 1;
116        if cmd & 0x80 != 0 {
117            // copy：从 base 复制 [offset, size]
118            let mut offset = 0usize;
119            for i in 0..4 {
120                if cmd & (1 << i) != 0 { offset |= (read_u8(delta, &mut pos)? as usize) << (8 * i); }
121            }
122            let mut size = 0usize;
123            for i in 0..3 {
124                if cmd & (1 << (4 + i)) != 0 { size |= (read_u8(delta, &mut pos)? as usize) << (8 * i); }
125            }
126            if size == 0 { size = 0x10000; } // 约定：size 字段为 0 表示 65536
127            let end = offset.checked_add(size).ok_or_else(|| bad("delta copy 越界"))?;
128            out.extend_from_slice(base.get(offset..end).ok_or_else(|| bad("delta copy 越界"))?);
129        } else if cmd != 0 {
130            // insert：把 delta 接下来的 cmd 字节当字面量
131            let n = cmd as usize;
132            let chunk = delta.get(pos..pos + n).ok_or_else(|| eof("delta insert 越界"))?;
133            out.extend_from_slice(chunk);
134            pos += n;
135        } else {
136            return Err(bad("delta 指令 0x00 非法"));
137        }
138    }
139    if out.len() != target_size {
140        return Err(bad("delta 重建后大小与声明不符"));
141    }
142    Ok(out)
143}

145fn object_sha(kind: Kind, content: &[u8]) -> [u8; 20] {
146    let mut h = Sha1::new();
147    h.update(format!("{} {}\0", kind.name(), content.len()).as_bytes()); // 关键前缀
148    h.update(content);
149    h.finalize().into()
150}

159/// thin-pack 版：ref-delta 的基对象允许**不在本 pack 内**，此时去 `git_dir`
160/// 的松散对象库里找并“增厚”。git fetch 的增量传输会产生这种包——服务器把新
161/// 对象 delta 在“客户端已有的旧对象”上，从而只发很小的差异。
162pub fn parse_pack_thin(data: &[u8], git_dir: &Path) -> io::Result<Vec<PackedObject>> {
163    parse_pack_inner(data, Some(git_dir))

166/// `git_dir` 为 `Some` 时支持 thin-pack（外部基对象从本地对象库解析）。
167//output是一系列的对象。
168fn parse_pack_inner(data: &[u8], git_dir: Option<&Path>) -> io::Result<Vec<PackedObject>> {
169    if data.len() < 12 + 20 {
170        return Err(bad("pack 太短"));
171    }
172    if &data[0..4] != b"PACK" {
173        return Err(bad("魔数不是 PACK"));
174    }
175    let version = u32::from_be_bytes(data[4..8].try_into().unwrap());
176    if version != 2 {
177        return Err(bad(format!("只支持版本 2，收到 {version}")));
178    }
179    let count = u32::from_be_bytes(data[8..12].try_into().unwrap()) as usize;
180
181    let mut pos = 12;
182    let mut objects: Vec<PackedObject> = Vec::with_capacity(count);
183    let mut by_offset: HashMap<usize, usize> = HashMap::new(); // pack偏移 → objects 下标
184    let mut by_sha: HashMap<[u8; 20], usize> = HashMap::new();
185    // thin-pack 外部基对象缓存：sha → (kind, 解压内容)，避免同一基重复读盘
186    let mut ext_cache: HashMap<[u8; 20], (Kind, Vec<u8>)> = HashMap::new();
187
188    for _ in 0..count {
189        let obj_offset = pos;
190        let (typ, size) = read_type_and_size(data, &mut pos)?;
191
192        let (kind, content) = match typ {
193            1 | 2 | 3 | 4 => {
194                let (content, used) = inflate(&data[pos..], size)?;
195                pos += used;
196                (Kind::from_type(typ).unwrap(), content)
197            }
198            6 => {
199                // ofs-delta：基在本 pack 更前面
200                let back = read_ofs_offset(data, &mut pos)?;
201                let base_off = obj_offset.checked_sub(back).ok_or_else(|| bad("ofs-delta 偏移越界"))?;
202                let (delta, used) = inflate(&data[pos..], size)?;
203                pos += used;
204                let &bi = by_offset.get(&base_off).ok_or_else(|| bad("ofs-delta 基对象未找到"))?;
205                let base = &objects[bi];
206                (base.kind, apply_delta(&base.data, &delta)?)
207            }
208            7 => {
209                // ref-delta：基用 20 字节 sha 指定
210                let base_sha: [u8; 20] = data
211                    .get(pos..pos + 20).ok_or_else(|| eof("ref-delta 缺基 sha"))?
212                    .try_into().unwrap();
213                pos += 20;
214                let (delta, used) = inflate(&data[pos..], size)?;
215                pos += used;
216                if let Some(&bi) = by_sha.get(&base_sha) {
217                    // 基就在本 pack 内
218                    let base = &objects[bi];
219                    (base.kind, apply_delta(&base.data, &delta)?)
220                } else {
221                    // thin-pack：基是“客户端已有”的外部对象，去本地对象库找
222                    let git_dir = git_dir.ok_or_else(|| {
223                        bad("ref-delta 基对象不在本 pack（thin pack，未提供本地对象库）")
224                    })?;
225                    if !ext_cache.contains_key(&base_sha) {
226                        let loaded = read_loose_base(git_dir, &base_sha)?.ok_or_else(|| {
227                            bad("ref-delta 外部基对象在本地对象库也找不到")
228                        })?;
229                        ext_cache.insert(base_sha, loaded);
230                    }
231                    let (bk, bd) = ext_cache.get(&base_sha).unwrap();
232                    (*bk, apply_delta(bd, &delta)?)
233                }
234            }
235            other => return Err(bad(format!("未知对象类型 {other}"))),
236        };
237
238        let sha = object_sha(kind, &content);
239        let idx = objects.len();
240        by_offset.insert(obj_offset, idx);
241        by_sha.insert(sha, idx);
242        objects.push(PackedObject { sha, kind, data: content });
243    }
244
245    // 尾部 20 字节：对前面所有字节求 SHA-1
246    let body = &data[..data.len() - 20];
247    let trailer = &data[data.len() - 20..];
248    let mut h = Sha1::new();
249    h.update(body);
250    let computed: [u8; 20] = h.finalize().into();
251    if computed != trailer {
252        return Err(bad("packfile 尾部 SHA-1 校验失败"));
253    }
254
255    Ok(objects)
256}

258/// 从本地对象库读一个松散对象，返回 `(kind, 解压后的内容payload)`（不含 `<type> <size>\0` 头）。
259/// 文件不存在返回 `Ok(None)`。供 thin-pack 的 ref-delta 解析外部基对象用。
260fn read_loose_base(git_dir: &Path, sha: &[u8; 20]) -> io::Result<Option<(Kind, Vec<u8>)>> {
261    let hex = to_hex(sha);
262    let path = git_dir.join("objects").join(&hex[0..2]).join(&hex[2..]);
263    let bytes = match fs::read(&path) {
264        Ok(b) => b,
265        Err(e) if e.kind() == io::ErrorKind::NotFound => return Ok(None),
266        Err(e) => return Err(e),
267    };
268    // loose 对象 = zlib( "<type> <size>\0" + content )
269    let mut dec = ZlibDecoder::new(&bytes[..]);
270    let mut raw = Vec::new();
271    dec.read_to_end(&mut raw)?;
272    let sp = raw.iter().position(|&b| b == b' ').ok_or_else(|| bad("loose 对象缺类型空格"))?;
273    let nul = raw.iter().position(|&b| b == 0).ok_or_else(|| bad("loose 对象缺 NUL"))?;
274    let kind = match &raw[..sp] {
275        b"commit" => Kind::Commit,
276        b"tree" => Kind::Tree,
277        b"blob" => Kind::Blob,
278        b"tag" => Kind::Tag,
279        other => return Err(bad(format!("未知 loose 类型 {}", String::from_utf8_lossy(other)))),
280    };
281    let content = raw[nul + 1..].to_vec();
282    Ok(Some((kind, content)))
283}

285/// 公开入口：按 40 位 hex 读本地松散对象，返回 `(kind, 解压后的内容payload)`。
286/// 对象不存在则报错。push 遍历对象图、取要发送的对象内容时用。
287pub fn read_loose_object_by_hex(git_abs: &Path, hex: &str) -> io::Result<(Kind, Vec<u8>)> {
288    let sha = hex_to_sha(hex)?;
289    read_loose_base(git_abs, &sha)?.ok_or_else(|| bad(format!("对象不在本地对象库: {hex}")))
290}

294/// 写一个对象条目的变长头（类型 + 解压后大小），是 `read_type_and_size` 的逆。
295fn write_pack_obj_header(out: &mut Vec<u8>, kind: Kind, size: usize) {
296    let t: u8 = match kind { Kind::Commit => 1, Kind::Tree => 2, Kind::Blob => 3, Kind::Tag => 4 };
297    let mut b = (t << 4) | ((size as u8) & 0x0f); // bit6-4=类型，低 4 位=size 低 4 bit
298    let mut size = size >> 4;
299    if size != 0 { b |= 0x80; }
300    out.push(b);
301    while size != 0 {
302        let mut c = (size & 0x7f) as u8;
303        size >>= 7;
304        if size != 0 { c |= 0x80; }
305        out.push(c);
306    }
307}

309/// 把若干 `(类型, 解压内容)` 组装成 pack v2 字节流：全部写成完整对象（不做 delta，
310/// 因此非 thin、自洽，可被 `parse_pack` / `git index-pack` 直接验证）。末尾附 20 字节 SHA-1。
311pub fn build_pack(objects: &[(Kind, Vec<u8>)]) -> io::Result<Vec<u8>> {
312    let mut out = Vec::new();
313    out.extend_from_slice(b"PACK");
314    out.extend_from_slice(&2u32.to_be_bytes());
315    out.extend_from_slice(&(objects.len() as u32).to_be_bytes());
316    for (kind, content) in objects {
317        write_pack_obj_header(&mut out, *kind, content.len());
318        let mut enc = ZlibEncoder::new(Vec::new(), Compression::default());
319        enc.write_all(content)?;
320        out.extend_from_slice(&enc.finish()?);
321    }
322    // 尾部 = 对前面所有字节求 SHA-1（与 parse_pack 的校验对应）
323    let mut h = Sha1::new();
324    h.update(&out);
325    let digest: [u8; 20] = h.finalize().into();
326    out.extend_from_slice(&digest);
327    Ok(out)
328}

368fn hex_to_sha(h: &str) -> io::Result<[u8; 20]> {
369    if h.len() != 40 {
370        return Err(err("sha 不是 40 位"));
371    }
372    let mut out = [0u8; 20];
373    for i in 0..20 {
374        out[i] = u8::from_str_radix(&h[2 * i..2 * i + 2], 16).map_err(|_| err("sha 非十六进制"))?;
375    }
376    Ok(out)
377}

381pub fn write_loose_object(git_dir: &Path, obj: &PackedObject) -> io::Result<()> {
382    let hex = to_hex(&obj.sha);
383    let dir = git_dir.join("objects").join(&hex[0..2]);
384    let path = dir.join(&hex[2..]);
385    if path.exists() {
386        return Ok(()); // 对象不可变，已存在就跳过
387    }
388    fs::create_dir_all(&dir)?;
389    // 内容 = zlib( "<type> <size>\0" + data )，正是第 6 步算 sha 哈希的那一坨
390    let mut raw = format!("{} {}\0", obj.kind.name(), obj.data.len()).into_bytes();
391    raw.extend_from_slice(&obj.data);
392    let mut enc = ZlibEncoder::new(Vec::new(), Compression::default());
393    enc.write_all(&raw)?;
394    fs::write(&path, enc.finish()?)
395}

399fn write_refs_and_head(git_dir: &Path, adv: &RefAdvertisement) -> io::Result<()> {
400    for r in &adv.refs {
401        if r.name == "HEAD" || r.name.ends_with("^{}") {
402            continue; // HEAD 单独写；peeled 行不写成 ref
403        }
404        let path = git_dir.join(&r.name); // 如 .git/refs/heads/main
405        if let Some(parent) = path.parent() {
406            fs::create_dir_all(parent)?;
407        }
408        fs::write(&path, format!("{}\n", r.sha))?; // 内容就是 40 位 sha
409    }
410    let head = adv.head_target.clone().unwrap_or_else(|| "refs/heads/master".to_string());
411    fs::write(git_dir.join("HEAD"), format!("ref: {head}\n")) // 符号引用
412}

416struct TreeEntry {
417    mode: u32,
418    name: String,
419    sha: [u8; 20],
420}
421 
422/// 解析 tree 的二进制：重复的 "<mode> <name>\0<20字节sha>"。
423fn parse_tree(tree: &[u8]) -> io::Result<Vec<TreeEntry>> {
424    let mut entries = Vec::new();
425    let mut pos = 0;
426    while pos < tree.len() {
427        let sp = tree[pos..].iter().position(|&b| b == b' ').ok_or_else(|| err("tree 缺空格"))? + pos;
428        let mode = u32::from_str_radix(
429            std::str::from_utf8(&tree[pos..sp]).map_err(|_| err("mode 非 utf8"))?,
430            8,
431        ).map_err(|_| err("mode 非八进制"))?;
432        let nul = tree[sp + 1..].iter().position(|&b| b == 0).ok_or_else(|| err("tree 缺 NUL"))? + sp + 1;
433        let name = String::from_utf8_lossy(&tree[sp + 1..nul]).into_owned();
434        let sha: [u8; 20] = tree.get(nul + 1..nul + 21).ok_or_else(|| err("tree sha 不足"))?
435            .try_into().unwrap();
436        pos = nul + 21;
437        entries.push(TreeEntry { mode, name, sha });
438    }
439    Ok(entries)
440}

442/// 从 commit 文本里取根 tree 的 sha（"tree <40hex>" 那行）。
443fn commit_tree_sha(commit: &[u8]) -> io::Result<[u8; 20]> {
444    let text = std::str::from_utf8(commit).map_err(|_| err("commit 非 utf8"))?;
445    for line in text.lines() {
446        if line.is_empty() {
447            break; // header 结束
448        }
449        if let Some(h) = line.strip_prefix("tree ") {
450            return hex_to_sha(h.trim());
451        }
452    }
453    Err(err("commit 里没有 tree 行"))
454}

456fn checkout_tree(
457    tree_sha: &[u8; 20],
458    dir: &Path,
459    by_sha: &HashMap<[u8; 20], &PackedObject>,
460) -> io::Result<()> {
461    let tree = by_sha.get(tree_sha).ok_or_else(|| err("tree 对象缺失"))?;
462    if tree.kind != Kind::Tree {
463        return Err(err("期望 tree 类型"));
464    }
465    for e in parse_tree(&tree.data)? {
466        let path = dir.join(&e.name);
467        match e.mode {
468            0o40000 => {
469                // 子目录 → 建目录、递归
470                fs::create_dir_all(&path)?;
471                checkout_tree(&e.sha, &path, by_sha)?;
472            }
473            0o160000 => { /* gitlink 子模块，先跳过 */ }
474            0o120000 => {
475                // 符号链接：blob 内容就是链接目标
476                let blob = by_sha.get(&e.sha).ok_or_else(|| err("symlink blob 缺失"))?;
477                #[cfg(unix)]
478                {
479                    use std::os::unix::ffi::OsStrExt;
480                    let _ = fs::remove_file(&path);
481                    std::os::unix::fs::symlink(std::ffi::OsStr::from_bytes(&blob.data), &path)?;
482                }
483                #[cfg(not(unix))]
484                fs::write(&path, &blob.data)?;
485            }
486            _ => {
487                // 100644 / 100755 普通文件
488                let blob = by_sha.get(&e.sha).ok_or_else(|| err("blob 缺失"))?;
489                fs::write(&path, &blob.data)?;
490                #[cfg(unix)]
491                if e.mode & 0o111 != 0 {
492                    use std::os::unix::fs::PermissionsExt;
493                    let mut perm = fs::metadata(&path)?.permissions();
494                    perm.set_mode(0o755);
495                    fs::set_permissions(&path, perm)?;
496                }
497            }
498        }
499    }
500    Ok(())
501}

503// ─────────── 顶层：把内存里的对象 + 广告，落成磁盘上的仓库 ───────────
504 
505pub fn clone_to_disk(
506    work_dir: &Path,
507    meta_dir: &str, // 实际用 META_DIR(".gift")；测试传 ".git" 以便直接用 git 验
508    objects: &[PackedObject],
509    adv: &RefAdvertisement,
510) -> io::Result<()> {
511    let git_dir = work_dir.join(meta_dir); // 内部结构与 .git 完全相同
512 
513    // 0) .git 骨架
514    fs::create_dir_all(git_dir.join("objects"))?;
515    fs::create_dir_all(git_dir.join("refs/heads"))?;
516    fs::create_dir_all(git_dir.join("refs/tags"))?;
517 
518    // 1) 对象落盘
519    for o in objects {
520        write_loose_object(&git_dir, o)?;
521    }
522 
523    // 2) refs + HEAD
524    write_refs_and_head(&git_dir, adv)?;
525 
526    // 3) checkout 工作区
527    let by_sha: HashMap<[u8; 20], &PackedObject> = objects.iter().map(|o| (o.sha, o)).collect();
528    if let Some(target) = adv.head_target.as_deref().and_then(|t| adv.sha_of(t)) {
529        let commit_sha = hex_to_sha(target)?;
530        let commit = by_sha.get(&commit_sha).ok_or_else(|| err("HEAD commit 缺失"))?;
531        if commit.kind != Kind::Commit {
532            return Err(err("HEAD 指向的不是 commit"));
533        }
534        let tree_sha = commit_tree_sha(&commit.data)?;
535        checkout_tree(&tree_sha, work_dir, &by_sha)?;
536    }
537    Ok(())
538}

540/// 像 git 那样，从远程 URL 推导出本地目录名。
541///
542/// 规则：去掉末尾的 `/`，取最后一段路径（`/` 或 scp 风格的 `:` 之后），
543/// 再去掉末尾的 `.git` 后缀。例如：
544/// - `https://github.com/foo/bar.git` → `bar`
545/// - `git@github.com:foo/bar.git`     → `bar`
546/// - `https://example.com/baz/`       → `baz`
547pub fn dir_name_from_url(url: &str) -> Result<String, anyhow::Error> {
548    let trimmed = url.trim_end_matches('/');
549    let last = trimmed
550        .rsplit(|c| c == '/' || c == ':')
551        .next()
552        .unwrap_or("");
553    let name = last.strip_suffix(".git").unwrap_or(last);
554    if name.is_empty() {
555        anyhow::bail!("无法从 URL 推导出目录名: {url}");
556    }
557    Ok(name.to_string())
558}
559
560pub fn clone(url: &str, dir: &str) -> Result<(), anyhow::Error> {
561    let work_dir = Path::new(dir);
562    std::fs::create_dir_all(work_dir)?;
563
564    let adv     = discover_refs(url)?;        // 第3步：RefAdvertisement（refs + HEAD）
565    let pack    = fetch_pack(url, &adv, &[])?; // 第4+5步：发 want（无 have）、解复用 → .pack 字节
566    let objects = parse_pack(&pack)?;         // 第6步：解析出 Vec<PackedObject>
567    clone_to_disk(work_dir, META_DIR, &objects, &adv)?; // 第7步：落盘 + checkout
568
569    Ok(())
570}
